#Scripted effect to release Estonia (R09)
pfp_release_estonia = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			812 = { add_contested_owner = R09 } #Harju
			13 = { add_contested_owner = R09 } #Parnu
			811 = { add_contested_owner = R09 } #Saaremaa
			813 = { add_contested_owner = R09 } #Virumaa
			191 = { add_contested_owner = R09 } #Tartu

			#Add claims to formable puppet
			812 = { add_claim_by = R09 } #Harju
			13 = { add_claim_by = R09 } #Parnu
			811 = { add_claim_by = R09 } #Saaremaa
			813 = { add_claim_by = R09 } #Virumaa
			191 = { add_claim_by = R09 } #Tartu

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = R09 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = R09
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = R09
				}
			}

			#Set custom focus tree to combat resistance
			R09 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			812 = { add_core_of = R09 } #Harju
			13 = { add_core_of = R09 } #Parnu
			811 = { add_core_of = R09 } #Saaremaa
			813 = { add_core_of = R09 } #Virumaa
			191 = { add_core_of = R09 } #Tartu

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = R09 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = R09
					transfer_state_to = R09
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = R09
					set_state_controller_to = R09
				}
			}

			R09 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

#Scripted effect to release North Sudan (Q31)
pfp_release_north_sudan = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			883 = { add_contested_owner = Q31 } #Kassala
			551 = { add_contested_owner = Q31 } #Khartoum
			767 = { add_contested_owner = Q31 } #North Darfur
			886 = { add_contested_owner = Q31 } #Blue Nile
			549 = { add_contested_owner = Q31 } #Kurdufan
			887 = { add_contested_owner = Q31 } #South Darfur
			884 = { add_contested_owner = Q31 } #Upper Nile
			885 = { add_contested_owner = Q31 } #Bahr al Ghazal

			#Add claims to formable puppet
			883 = { add_claim_by = Q31 } #Kassala
			551 = { add_claim_by = Q31 } #Khartoum
			767 = { add_claim_by = Q31 } #North Darfur
			886 = { add_claim_by = Q31 } #Blue Nile
			549 = { add_claim_by = Q31 } #Kurdufan
			887 = { add_claim_by = Q31 } #South Darfur
			884 = { add_claim_by = Q31 } #Upper Nile
			885 = { add_claim_by = Q31 } #Bahr al Ghazal

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = Q31 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q31
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to = Q31
				}
			}

			#Set custom focus tree to combat resistance
			Q31 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			883 = { add_core_of = Q31 } #Kassala
			551 = { add_core_of = Q31 } #Khartoum
			767 = { add_core_of = Q31 } #North Darfur
			886 = { add_core_of = Q31 } #Blue Nile
			549 = { add_core_of = Q31 } #Kurdufan
			887 = { add_core_of = Q31 } #South Darfur
			884 = { add_core_of = Q31 } #Upper Nile
			885 = { add_core_of = Q31 } #Bahr al Ghazal

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = Q31 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q31
					transfer_state_to = Q31
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q31
					set_state_controller_to = Q31
				}
			}

			Q31 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

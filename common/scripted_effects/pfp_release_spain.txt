	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			41 = { add_contested_owner = Q80 } #Madrid
			791 = { add_contested_owner = Q80 } #Valladolid
			174 = { add_contested_owner = Q80 } #León
			790 = { add_contested_owner = Q80 } #Asturias
			176 = { add_contested_owner = Q80 } #Burgos
			793 = { add_contested_owner = Q80 } #Guadalajara
			788 = { add_contested_owner = Q80 } #Salamanca
			166 = { add_contested_owner = Q80 } #Western Aragón
			794 = { add_contested_owner = Q80 } #Eastern Aragón
			167 = { add_contested_owner = Q80 } #Valencia
			168 = { add_contested_owner = Q80 } #Murcia
			175 = { add_contested_owner = Q80 } #Ciudad Real
			170 = { add_contested_owner = Q80 } #Extremadura
			169 = { add_contested_owner = Q80 } #Sevilla
			789 = { add_contested_owner = Q80 } #Córdoba
			173 = { add_contested_owner = Q80 } #Granada
			165 = { add_contested_owner = Q80 } #Catalunya
			172 = { add_contested_owner = Q80 } #Navarra
			792 = { add_contested_owner = Q80 } #País Vasco
			171 = { add_contested_owner = Q80 } #Galicia

			#Add claims to formable puppet
			41 = { add_claim_by = Q80 } #Madrid
			791 = { add_claim_by = Q80 } #Valladolid
			174 = { add_claim_by = Q80 } #León
			790 = { add_claim_by = Q80 } #Asturias
			176 = { add_claim_by = Q80 } #Burgos
			793 = { add_claim_by = Q80 } #Guadalajara
			788 = { add_claim_by = Q80 } #Salamanca
			166 = { add_claim_by = Q80 } #Western Aragón
			794 = { add_claim_by = Q80 } #Eastern Aragón
			167 = { add_claim_by = Q80 } #Valencia
			168 = { add_claim_by = Q80 } #Murcia
			175 = { add_claim_by = Q80 } #Ciudad Real
			170 = { add_claim_by = Q80 } #Extremadura
			169 = { add_claim_by = Q80 } #Sevilla
			789 = { add_claim_by = Q80 } #Córdoba
			173 = { add_claim_by = Q80 } #Granada
			165 = { add_claim_by = Q80 } #Catalunya
			172 = { add_claim_by = Q80 } #Navarra
			792 = { add_claim_by = Q80 } #País Vasco
			171 = { add_claim_by = Q80 } #Galicia

			#Set formable puppet as controller of contested states
			every_state = {
				limit = { has_contested_owner = Q80 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q80
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = Q80
				}
			}

			#Set custom focus tree to combat resistance
			Q80 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			41 = { add_core_of = Q80 } #Madrid
			791 = { add_core_of = Q80 } #Valladolid
			174 = { add_core_of = Q80 } #León
			790 = { add_core_of = Q80 } #Asturias
			176 = { add_core_of = Q80 } #Burgos
			793 = { add_core_of = Q80 } #Guadalajara
			788 = { add_core_of = Q80 } #Salamanca
			166 = { add_core_of = Q80 } #Western Aragón
			794 = { add_core_of = Q80 } #Eastern Aragón
			167 = { add_core_of = Q80 } #Valencia
			168 = { add_core_of = Q80 } #Murcia
			175 = { add_core_of = Q80 } #Ciudad Real
			170 = { add_core_of = Q80 } #Extremadura
			169 = { add_core_of = Q80 } #Sevilla
			789 = { add_core_of = Q80 } #Córdoba
			173 = { add_core_of = Q80 } #Granada
			165 = { add_core_of = Q80 } #Catalunya
			172 = { add_core_of = Q80 } #Navarra
			792 = { add_core_of = Q80 } #País Vasco
			171 = { add_core_of = Q80 } #Galicia

			#Set formable puppet as owner and controller of core states
			every_state = {
				limit = { is_core_of = Q80 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q80
					transfer_state_to = Q80
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q80
					set_state_controller_to = Q80
				}
			}

			Q80 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

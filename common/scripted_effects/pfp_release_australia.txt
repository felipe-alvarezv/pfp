#Scripted effect to release Australia (R19)
pfp_release_australia = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			285 = { add_contested_owner = R19 } #New South Wales
			517 = { add_contested_owner = R19 } #Victoria
			521 = { add_contested_owner = R19 } #Queensland
			873 = { add_contested_owner = R19 } #Channel Country
			519 = { add_contested_owner = R19 } #South Australia
			1075 = { add_contested_owner = R19 } #West Queensland
			872 = { add_contested_owner = R19 } #North Queensland
			674 = { add_contested_owner = R19 } #Central Australia
			520 = { add_contested_owner = R19 } #Northern Territories
			871 = { add_contested_owner = R19 } #Esperence-Goldfields
			1074 = { add_contested_owner = R19 } #Gibson Desert
			870 = { add_contested_owner = R19 } #Pilbara-Kimberley
			522 = { add_contested_owner = R19 } #Western Australia

			#Add claims to formable puppet
			285 = { add_claim_by = R19 } #New South Wales
			517 = { add_claim_by = R19 } #Victoria
			521 = { add_claim_by = R19 } #Queensland
			873 = { add_claim_by = R19 } #Channel Country
			519 = { add_claim_by = R19 } #South Australia
			1075 = { add_claim_by = R19 } #West Queensland
			872 = { add_claim_by = R19 } #North Queensland
			674 = { add_claim_by = R19 } #Central Australia
			520 = { add_claim_by = R19 } #Northern Territories
			871 = { add_claim_by = R19 } #Esperence-Goldfields
			1074 = { add_claim_by = R19 } #Gibson Desert
			870 = { add_claim_by = R19 } #Pilbara-Kimberley
			522 = { add_claim_by = R19 } #Western Australia

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = R19 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = R19
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = R19
				}
			}

			#Set custom focus tree to combat resistance
			R19 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			285 = { add_core_of = R19 } #New South Wales
			517 = { add_core_of = R19 } #Victoria
			521 = { add_core_of = R19 } #Queensland
			873 = { add_core_of = R19 } #Channel Country
			519 = { add_core_of = R19 } #South Australia
			1075 = { add_core_of = R19 } #West Queensland
			872 = { add_core_of = R19 } #North Queensland
			674 = { add_core_of = R19 } #Central Australia
			520 = { add_core_of = R19 } #Northern Territories
			871 = { add_core_of = R19 } #Esperence-Goldfields
			1074 = { add_core_of = R19 } #Gibson Desert
			870 = { add_core_of = R19 } #Pilbara-Kimberley
			522 = { add_core_of = R19 } #Western Australia

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = R19 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = R19
					transfer_state_to = R19
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = R19
					set_state_controller_to = R19
				}
			}

			R19 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

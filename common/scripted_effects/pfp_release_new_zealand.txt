#Scripted effect to release New Zealand (R20)
pfp_release_new_zealand = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			284 = { add_contested_owner = R20 } #Wellington
			1079 = { add_contested_owner = R20 } #Auckland
			1080 = { add_contested_owner = R20 } #Marlborough
			723 = { add_contested_owner = R20 } #Canterbury
			1081 = { add_contested_owner = R20 } #Otago

			#Add claims to formable puppet
			284 = { add_claim_by = R20 } #Wellington
			1079 = { add_claim_by = R20 } #Auckland
			1080 = { add_claim_by = R20 } #Marlborough
			723 = { add_claim_by = R20 } #Canterbury
			1081 = { add_claim_by = R20 } #Otago

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = R20 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = R20
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = R20
				}
			}

			#Set custom focus tree to combat resistance
			R20 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			284 = { add_core_of = R20 } #Wellington
			1079 = { add_core_of = R20 } #Auckland
			1080 = { add_core_of = R20 } #Marlborough
			723 = { add_core_of = R20 } #Canterbury
			1081 = { add_core_of = R20 } #Otago

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = R20 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = R20
					transfer_state_to = R20
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = R20
					set_state_controller_to = R20
				}
			}

			R20 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

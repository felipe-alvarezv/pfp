#Scripted effect to release Ethiopia (Q55)
pfp_release_ethiopia = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			271 = { add_contested_owner = Q55 } #Shewa
			842 = { add_contested_owner = Q55 } #Tigray
			841 = { add_contested_owner = Q55 } #Begemder
			843 = { add_contested_owner = Q55 } #Wello
			840 = { add_contested_owner = Q55 } #Gojjam
			839 = { add_contested_owner = Q55 } #Welega
			838 = { add_contested_owner = Q55 } #Illubabor-Kaffa
			835 = { add_contested_owner = Q55 } #Hararghe
			836 = { add_contested_owner = Q55 } #Bale
			837 = { add_contested_owner = Q55 } #Sidamo

			#Add claims to formable puppet
			271 = { add_claim_by = Q55 } #Shewa
			842 = { add_claim_by = Q55 } #Tigray
			841 = { add_claim_by = Q55 } #Begemder
			843 = { add_claim_by = Q55 } #Wello
			840 = { add_claim_by = Q55 } #Gojjam
			839 = { add_claim_by = Q55 } #Welega
			838 = { add_claim_by = Q55 } #Illubabor-Kaffa
			835 = { add_claim_by = Q55 } #Hararghe
			836 = { add_claim_by = Q55 } #Bale
			837 = { add_claim_by = Q55 } #Sidamo

			#Set formable puppet as controller of contested states
			every_state = {
				limit = { has_contested_owner = Q55 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q55
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to = Q55
				}
			}

			#Set custom focus tree to combat resistance
			Q55 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			271 = { add_core_of = Q55 } #Shewa
			842 = { add_core_of = Q55 } #Tigray
			841 = { add_core_of = Q55 } #Begemder
			843 = { add_core_of = Q55 } #Wello
			840 = { add_core_of = Q55 } #Gojjam
			839 = { add_core_of = Q55 } #Welega
			838 = { add_core_of = Q55 } #Illubabor-Kaffa
			835 = { add_core_of = Q55 } #Hararghe
			836 = { add_core_of = Q55 } #Bale
			837 = { add_core_of = Q55 } #Sidamo

			#Set formable puppet as owner and controller of core states
			every_state = {
				limit = { is_core_of = Q55 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q55
					transfer_state_to = Q55
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q55
					set_state_controller_to = Q55
				}
			}

			Q55 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

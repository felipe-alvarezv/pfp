#Scripted effect to release Portugal (Q81)
pfp_release_portugal = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			112 = { add_contested_owner = Q81 } #Lisbon
			180 = { add_contested_owner = Q81 } #Porto
			181 = { add_contested_owner = Q81 } #Guarda
			795 = { add_contested_owner = Q81 } #Santarém
			179 = { add_contested_owner = Q81 } #Beja

			#Add claims to formable puppet
			112 = { add_claim_by = Q81 } #Lisbon
			180 = { add_claim_by = Q81 } #Porto
			181 = { add_claim_by = Q81 } #Guarda
			795 = { add_claim_by = Q81 } #Santarém
			179 = { add_claim_by = Q81 } #Beja

			#Set formable puppet as controller of contested states
			every_state = {
				limit = { has_contested_owner = Q81 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q81
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = Q81
				}
			}

			#Set custom focus tree to combat resistance
			Q81 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			112 = { add_core_of = Q81 } #Lisbon
			180 = { add_core_of = Q81 } #Porto
			181 = { add_core_of = Q81 } #Guarda
			795 = { add_core_of = Q81 } #Santarém
			179 = { add_core_of = Q81 } #Beja

			#Set formable puppet as owner and controller of core states
			every_state = {
				limit = { is_core_of = Q81 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q81
					transfer_state_to = Q81
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q81
					set_state_controller_to = Q81
				}
			}

			Q81 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

#Scripted effect to release South Korea (R39)
pfp_release_south_korea = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			525 = { add_contested_owner = R39 } #Gyeonggi
			1029 = { add_contested_owner = R39 } #Gangwon
			1031 = { add_contested_owner = R39 } #Chungcheong-Jeolla
			1030 = { add_contested_owner = R39 } #Gyeongsang

			#Add claims to formable puppet
			525 = { add_claim_by = R39 } #Gyeonggi
			1029 = { add_claim_by = R39 } #Gangwon
			1031 = { add_claim_by = R39 } #Chungcheong-Jeolla
			1030 = { add_claim_by = R39 } #Gyeongsang

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = R39 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = R39
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = R39
				}
			}

			#Set custom focus tree to combat resistance
			R39 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			525 = { add_core_of = R39 } #Gyeonggi
			1029 = { add_core_of = R39 } #Gangwon
			1031 = { add_core_of = R39 } #Chungcheong-Jeolla
			1030 = { add_core_of = R39 } #Gyeongsang

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = R39 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = R39
					transfer_state_to = R39
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = R39
					set_state_controller_to = R39
				}
			}

			R39 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

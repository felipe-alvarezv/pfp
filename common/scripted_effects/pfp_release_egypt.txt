#Scripted effect to release Egypt (Q30)
pfp_release_egypt = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			452 = { add_contested_owner = Q30 } #Matrouh
			552 = { add_contested_owner = Q30 } #Western Desert
			447 = { add_contested_owner = Q30 } #Alexandria
			907 = { add_contested_owner = Q30 } #Cairo
			456 = { add_contested_owner = Q30 } #Aswan
			457 = { add_contested_owner = Q30 } #Eastern Desert

			#Add claims to formable puppet
			452 = { add_claim_by = Q30 } #Matrouh
			552 = { add_claim_by = Q30 } #Western Desert
			447 = { add_claim_by = Q30 } #Alexandria
			907 = { add_claim_by = Q30 } #Cairo
			456 = { add_claim_by = Q30 } #Aswan
			457 = { add_claim_by = Q30 } #Eastern Desert

			#Set formable puppet as controller of contested state
			every_state = {
				limit = { has_contested_owner = Q30 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q30
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to = Q30
				}
			}

			#Set custom focus tree to combat resistance
			Q30 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			452 = { add_core_of = Q30 } #Matrouh
			552 = { add_core_of = Q30 } #Western Desert
			447 = { add_core_of = Q30 } #Alexandria
			907 = { add_core_of = Q30 } #Cairo
			456 = { add_core_of = Q30 } #Aswan
			457 = { add_core_of = Q30 } #Eastern Desert

			#Set formable puppet as owner and controller of core state
			every_state = {
				limit = { is_core_of = Q30 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q30
					transfer_state_to = Q30
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q30
					set_state_controller_to = Q30
				}
			}

			Q30 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

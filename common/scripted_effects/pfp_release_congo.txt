#Scripted effect to release Congo (Q65)
pfp_release_congo = {
	hidden_effect = {
		#If resistance mechanics are enabled, formable puppets will contest and control its corresponding states.
		if = {
			limit = { has_global_flag = pfp_management_resistance_enabled }

			#Add contestion to formable puppet
			295 = { add_contested_owner = Q65 } #Léopoldville
			538 = { add_contested_owner = Q65 } #Coquilhatville
			718 = { add_contested_owner = Q65 } #Stanleyville
			888 = { add_contested_owner = Q65 } #Lusambo
			889 = { add_contested_owner = Q65 } #Elisabethville
			890 = { add_contested_owner = Q65 } #Costermansville

			#Add claims to formable puppet
			295 = { add_claim_by = Q65 } #Léopoldville
			538 = { add_claim_by = Q65 } #Coquilhatville
			718 = { add_claim_by = Q65 } #Stanleyville
			888 = { add_claim_by = Q65 } #Lusambo
			889 = { add_claim_by = Q65 } #Elisabethville
			890 = { add_claim_by = Q65 } #Costermansville

			#Set formable puppet as controller of contested states
			every_state = {
				limit = { has_contested_owner = Q65 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }
					transfer_state_to = Q65
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					set_state_controller_to  = Q65
				}
			}

			#Set custom focus tree to combat resistance
			Q65 = { load_focus_tree = pfp_formable_puppet_focus }
		}
		#If resistance mechanics are disabled, formable puppets will control and core its corresponding states.
		else = {
			#Add cores to formable puppet
			295 = { add_core_of = Q65 } #Léopoldville
			538 = { add_core_of = Q65 } #Coquilhatville
			718 = { add_core_of = Q65 } #Stanleyville
			888 = { add_core_of = Q65 } #Lusambo
			889 = { add_core_of = Q65 } #Elisabethville
			890 = { add_core_of = Q65 } #Costermansville

			#Set formable puppet as owner and controller of core states
			every_state = {
				limit = { is_core_of = Q65 }

				#If ROOT owns the state, transfer control and ownership to formable puppet
				if = {
					limit = { is_owned_by = ROOT }

					add_contested_owner = Q65
					transfer_state_to = Q65
				}

				#If ROOT does not own the state, transfer control to formable puppet
				else = {
					add_contested_owner = Q65
					set_state_controller_to = Q65
				}
			}

			Q65 = {
				remove_ideas = {
					pfp_puppet_resistance_idea_1
				}
			}
		}
	}
}

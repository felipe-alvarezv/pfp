###########################################################################
# MERGE DECISIONS - YUGOSLAVIA
# Allows the overlord to merge Yugoslav puppet states into a unified
# Yugoslavia.
#
# Decision 1: pfp_merge_yugoslavia
#   Requires Croatia (P86), Slovenia (P87), Bosnia-Herzegovina (P88),
#   and Serbia (P89) as PFP subjects.
#   The puppet with the most total factories + scaled manpower becomes
#   Yugoslavia (PFP_YUGOSLAVIA).
#
# Follow-up decisions (available after Yugoslavia is formed):
#   pfp_merge_yugoslavia_montenegro - Absorbs Montenegro (P90)
#   pfp_merge_yugoslavia_macedonia  - Absorbs Macedonia  (P98)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Croatia + Slovenia + Bosnia-Herzegovina + Serbia -> Yugoslavia
	# ----------------------------------------------------------------
	pfp_merge_yugoslavia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_yugoslavia_merged }
			OR = {
				P86 = { is_subject_of = ROOT }
				P87 = { is_subject_of = ROOT }
				P88 = { is_subject_of = ROOT }
				P89 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_yugoslavia_merged }
			}
			P86 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P87 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P88 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P89 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_form
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_annex
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories + (manpower / 100000)
				P86 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P87 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P88 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P89 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }

				# Initialize winner with P86, then linear challenge
				P86 = { save_global_event_target_as = pfp_yugoslavia_winner }
				if = { limit = { check_variable = { P87.pfp_fac > event_target:pfp_yugoslavia_winner.pfp_fac } } P87 = { save_global_event_target_as = pfp_yugoslavia_winner } }
				if = { limit = { check_variable = { P88.pfp_fac > event_target:pfp_yugoslavia_winner.pfp_fac } } P88 = { save_global_event_target_as = pfp_yugoslavia_winner } }
				if = { limit = { check_variable = { P89.pfp_fac > event_target:pfp_yugoslavia_winner.pfp_fac } } P89 = { save_global_event_target_as = pfp_yugoslavia_winner } }

				# Transfer state cores/ownership from non-winner subjects to winner
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P86 } } }
					every_state = { limit = { OR = { is_owned_by = P86 has_contested_owner = P86 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P86 remove_claim_by = P86 remove_contested_owner = P86 }
				}
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P87 } } }
					every_state = { limit = { OR = { is_owned_by = P87 has_contested_owner = P87 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P87 remove_claim_by = P87 remove_contested_owner = P87 }
				}
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P88 } } }
					every_state = { limit = { OR = { is_owned_by = P88 has_contested_owner = P88 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P88 remove_claim_by = P88 remove_contested_owner = P88 }
				}
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P89 } } }
					every_state = { limit = { OR = { is_owned_by = P89 has_contested_owner = P89 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P89 remove_claim_by = P89 remove_contested_owner = P89 }
				}

				# Transfer pfp_possible_cores arrays, annex non-winners, set tag and flag
				event_target:pfp_yugoslavia_winner = {
					if = { limit = { NOT = { tag = P86 } } for_each_loop = { array = P86.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P87 } } for_each_loop = { array = P87.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P88 } } for_each_loop = { array = P88.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P89 } } for_each_loop = { array = P89.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P86 } } annex_country = { target = P86 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P87 } } annex_country = { target = P87 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P88 } } annex_country = { target = P88 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P89 } } annex_country = { target = P89 transfer_troops = yes } }
					set_cosmetic_tag = PFP_YUGOSLAVIA
					set_country_flag = pfp_is_yugoslavia
				}

				clear_global_event_target = pfp_yugoslavia_winner
				set_global_flag = pfp_yugoslavia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Yugoslavia + Montenegro (P90) -> absorbed into Yugoslavia
	# ----------------------------------------------------------------
	pfp_merge_yugoslavia_montenegro = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_yugoslavia_merged
			NOT = { has_global_flag = pfp_yugoslavia_absorbed_montenegro }
			P90 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_montenegro_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_yugoslavia_absorbed_montenegro }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_yugoslavia }
			}
			P90 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_yugoslavia_montenegro_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_yugoslavia }
					save_global_event_target_as = pfp_yugoslavia_country
					for_each_loop = { array = P90.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
				}
				every_state = {
					limit = { OR = { is_owned_by = P90 has_contested_owner = P90 } }
					add_core_of = event_target:pfp_yugoslavia_country
					add_contested_owner = event_target:pfp_yugoslavia_country
					remove_core_of = P90
					remove_claim_by = P90
					remove_contested_owner = P90
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_yugoslavia }
					annex_country = { target = P90 transfer_troops = yes }
				}
				clear_global_event_target = pfp_yugoslavia_country
				set_global_flag = pfp_yugoslavia_absorbed_montenegro
			}
		}

		ai_will_do = { factor = 0 }
	}

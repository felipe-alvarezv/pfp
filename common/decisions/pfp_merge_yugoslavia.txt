###########################################################################
# MERGE DECISIONS - YUGOSLAVIA
# Allows the overlord to merge Yugoslav puppet states into a unified
# Yugoslavia.
#
# Decision 1: pfp_merge_yugoslavia
#   Requires Serbia (P89) and Montenegro (P90) as PFP subjects.
#   The puppet with the most total factories becomes Yugoslavia (PFP_YUGOSLAVIA).
#
# Follow-up decisions (available after Yugoslavia is formed):
#   pfp_merge_yugoslavia_macedonia          - Absorbs Macedonia          (P98)
#   pfp_merge_yugoslavia_slovenia           - Absorbs Slovenia           (P87)
#   pfp_merge_yugoslavia_croatia            - Absorbs Croatia            (P86)
#   pfp_merge_yugoslavia_bosnia_herzegovina - Absorbs Bosnia-Herzegovina (P88)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Serbia + Montenegro -> Yugoslavia
	# ----------------------------------------------------------------
	pfp_merge_yugoslavia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_yugoslavia_merged }
			OR = {
				P89 = { is_subject_of = ROOT }
				P90 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_yugoslavia_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_puppets_ready_tt
				P89 = { is_subject_of = ROOT }
				P90 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				P89 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				P90 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				P89 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				P90 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_form
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_annex
			custom_effect_tooltip = pfp_merge_yugoslavia_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories
				P89 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }
				P90 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }

				# Initialize winner with P89, then linear challenge
				P89 = { save_global_event_target_as = pfp_yugoslavia_winner }
				if = { limit = { check_variable = { P90.pfp_fac > event_target:pfp_yugoslavia_winner.pfp_fac } } P90 = { save_global_event_target_as = pfp_yugoslavia_winner } }

				# Transfer state cores/ownership from non-winner to winner
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P89 } } }
					every_state = { limit = { OR = { is_owned_by = P89 has_contested_owner = P89 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P89 remove_claim_by = P89 remove_contested_owner = P89 }
				}
				if = {
					limit = { NOT = { event_target:pfp_yugoslavia_winner = { tag = P90 } } }
					every_state = { limit = { OR = { is_owned_by = P90 has_contested_owner = P90 } } add_core_of = event_target:pfp_yugoslavia_winner add_contested_owner = event_target:pfp_yugoslavia_winner remove_core_of = P90 remove_claim_by = P90 remove_contested_owner = P90 }
				}

				# Transfer pfp_possible_cores arrays, annex non-winners, set tag and flag
				event_target:pfp_yugoslavia_winner = {
					if = { limit = { NOT = { tag = P89 } } for_each_loop = { array = P89.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P90 } } for_each_loop = { array = P90.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P89 } } annex_country = { target = P89 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P90 } } annex_country = { target = P90 transfer_troops = yes } }
					set_cosmetic_tag = PFP_YUGOSLAVIA
					set_country_flag = pfp_is_yugoslavia
				}

				clear_global_event_target = pfp_yugoslavia_winner
				set_global_flag = pfp_yugoslavia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Yugoslavia + Macedonia (P98) -> absorbed into Yugoslavia
	# ----------------------------------------------------------------
	pfp_merge_yugoslavia_macedonia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_yugoslavia_merged

			P98 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_macedonia_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_yugoslavia_absorbed_macedonia }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_yugoslavia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_yugoslavia }
			}
			P98 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
					has_autonomy_state = pfp_autonomy_military_occupation
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_yugoslavia_macedonia_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_yugoslavia }
					save_global_event_target_as = pfp_yugoslavia_country
					for_each_loop = { array = P98.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
				}
				every_state = {
					limit = { OR = { is_owned_by = P98 has_contested_owner = P98 } }
					add_core_of = event_target:pfp_yugoslavia_country
					add_contested_owner = event_target:pfp_yugoslavia_country
					remove_core_of = P98
					remove_claim_by = P98
					remove_contested_owner = P98
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_yugoslavia }
					annex_country = { target = P98 transfer_troops = yes }
				}
				clear_global_event_target = pfp_yugoslavia_country
				set_global_flag = pfp_yugoslavia_absorbed_macedonia
			}
		}

		ai_will_do = { factor = 0 }
	}

pfp_category = {
	pfp_djibouti = {
		allowed = {
			NOT = {
				original_tag = DJI #Djibouti
			}
		}
		available = {
			is_subject = no
			#Djibouti
			custom_trigger_tooltip = {
				tooltip = pfp_djibouti_required_states_french_somaliland
				has_full_control_of_state = 268 #French Somaliland
			}
		}
		visible = {
			has_full_control_of_state = 268 #French Somaliland
			NOT = { has_global_flag = pfp_djibouti_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 268 #French Somaliland
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_djibouti_transfer_capital
			custom_effect_tooltip = pfp_djibouti_puppet
			custom_effect_tooltip = pfp_djibouti_occupation_type
			custom_effect_tooltip = pfp_djibouti_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				268 = { transfer_state_to = Q57 } #French Somaliland

				#Formable puppet becomes a puppet of ROOT
				puppet = Q57

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q57 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_djibouti = yes

				#Transfer all technology from ROOT to formable puppet
				Q57 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_djibouti_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

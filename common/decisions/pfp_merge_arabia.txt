###########################################################################
# MERGE DECISIONS - ARABIA
# Allows the overlord to merge Arabian puppet states into a unified Arabia.
#
# Decision 1: pfp_merge_arabia
#   Requires Saudi Arabia (Q85), Yemen (Q83), and Oman (Q84) as PFP subjects.
#   The puppet with the most total factories becomes Arabia (PFP_ARABIA).
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Saudi Arabia + Yemen + Oman -> Arabia
	# ----------------------------------------------------------------
	pfp_merge_arabia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_arabia_merged }
			OR = {
				Q83 = { is_subject_of = ROOT }
				Q84 = { is_subject_of = ROOT }
				Q85 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_arabia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_arabia_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_arabia_puppets_ready_tt
				Q83 = { is_subject_of = ROOT }
				Q84 = { is_subject_of = ROOT }
				Q85 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				Q83 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q84 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q85 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				Q83 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q84 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q85 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_arabia_effect_form
			custom_effect_tooltip = pfp_merge_arabia_effect_annex
			custom_effect_tooltip = pfp_merge_arabia_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories
				Q83 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }
				Q84 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }
				Q85 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }

				# Initialize winner with Q85, challenge with Q83 and Q84
				Q85 = { save_global_event_target_as = pfp_arabia_winner }
				if = { limit = { check_variable = { Q83.pfp_fac > event_target:pfp_arabia_winner.pfp_fac } } Q83 = { save_global_event_target_as = pfp_arabia_winner } }
				if = { limit = { check_variable = { Q84.pfp_fac > event_target:pfp_arabia_winner.pfp_fac } } Q84 = { save_global_event_target_as = pfp_arabia_winner } }

				# Transfer state cores/ownership from non-winners to winner
				if = {
					limit = { NOT = { event_target:pfp_arabia_winner = { tag = Q83 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q83  has_contested_owner = Q83 } }
						add_core_of = event_target:pfp_arabia_winner
						add_contested_owner = event_target:pfp_arabia_winner
						remove_core_of = Q83
						remove_claim_by = Q83
						remove_contested_owner = Q83
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_arabia_winner = { tag = Q84 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q84  has_contested_owner = Q84 } }
						add_core_of = event_target:pfp_arabia_winner
						add_contested_owner = event_target:pfp_arabia_winner
						remove_core_of = Q84
						remove_claim_by = Q84
						remove_contested_owner = Q84
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_arabia_winner = { tag = Q85 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q85  has_contested_owner = Q85 } }
						add_core_of = event_target:pfp_arabia_winner
						add_contested_owner = event_target:pfp_arabia_winner
						remove_core_of = Q85
						remove_claim_by = Q85
						remove_contested_owner = Q85
					}
				}

				# Transfer pfp_possible_cores array, annex losers, set tag and flag
				event_target:pfp_arabia_winner = {
					if = { limit = { NOT = { tag = Q83 } } for_each_loop = { array = Q83.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q84 } } for_each_loop = { array = Q84.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q85 } } for_each_loop = { array = Q85.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q83 } } annex_country = { target = Q83 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q84 } } annex_country = { target = Q84 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q85 } } annex_country = { target = Q85 transfer_troops = yes } }
					set_cosmetic_tag = PFP_ARABIA
					set_country_flag = pfp_is_arabia
				}

				clear_global_event_target = pfp_arabia_winner
				set_global_flag = pfp_arabia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

}

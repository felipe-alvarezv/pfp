pfp_category = {
	pfp_belize = {
		allowed = {
			NOT = {
				original_tag = BLZ #Belize
			}
		}
		available = {
			is_subject = no
			#Belize
			custom_trigger_tooltip = {
				tooltip = pfp_belize_required_states_british_honduras
				has_full_control_of_state = 311 #British Honduras
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 311 #British Honduras
			}
			NOT = { has_global_flag = pfp_belize_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 311 #British Honduras
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_belize_transfer_capital
			custom_effect_tooltip = pfp_belize_puppet
			custom_effect_tooltip = pfp_belize_occupation_type
			custom_effect_tooltip = pfp_belize_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				311 = { transfer_state_to = R08 } #British Honduras

				#Formable puppet becomes a puppet of ROOT
				puppet = R08

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R08 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_belize = yes

				#Transfer all technology from ROOT to formable puppet
				R08 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_belize_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

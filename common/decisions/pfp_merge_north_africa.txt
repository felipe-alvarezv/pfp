###########################################################################
# MERGE DECISIONS - NORTH AFRICA
# Allows the overlord to merge North African puppet states into a unified
# North Africa.
#
# Decision 1: pfp_merge_north_africa
#   Requires all 5 core North African puppets as PFP subjects:
#   Morocco (Q26), Algeria (Q27), Tunisia (Q28), Libya (Q29), Egypt (Q30).
#   The puppet with the most total factories becomes North Africa.
#
# Follow-up decisions (available after North Africa is formed):
#   pfp_merge_north_africa_western_sahara - Absorbs Western Sahara (Q53)
#   pfp_merge_north_africa_rif            - Absorbs Rif (Q73)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Morocco (Q26) + Algeria (Q27) + Tunisia (Q28) +
	# Libya (Q29) + Egypt (Q30) -> North Africa
	# ----------------------------------------------------------------
	pfp_merge_north_africa = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_north_africa_merged }
			OR = {
				Q26 = { is_subject_of = ROOT }
				Q27 = { is_subject_of = ROOT }
				Q28 = { is_subject_of = ROOT }
				Q29 = { is_subject_of = ROOT }
				Q30 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_north_africa_not_yet_formed_tt
				NOT = { has_global_flag = pfp_north_africa_merged }
			}
			Q26 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			Q27 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			Q28 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			Q29 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			Q30 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_north_africa_effect_form
			custom_effect_tooltip = pfp_merge_north_africa_effect_annex
			custom_effect_tooltip = pfp_merge_north_africa_effect_cores
			hidden_effect = {
				# Q26 Morocco initializes as default winner.
				# Each subsequent tag wins if it strictly has more factories.
				Q26 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_north_africa_winner
				}
				Q27 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_north_africa_winner.pfp_fac } }
						save_global_event_target_as = pfp_north_africa_winner
					}
				}
				Q28 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_north_africa_winner.pfp_fac } }
						save_global_event_target_as = pfp_north_africa_winner
					}
				}
				Q29 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_north_africa_winner.pfp_fac } }
						save_global_event_target_as = pfp_north_africa_winner
					}
				}
				Q30 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_north_africa_winner.pfp_fac } }
						save_global_event_target_as = pfp_north_africa_winner
					}
				}

				# Transfer state cores/ownership from losers to winner
				if = {
					limit = { NOT = { event_target:pfp_north_africa_winner = { tag = Q26 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q26  has_contested_owner = Q26 } }
						add_core_of = event_target:pfp_north_africa_winner
						add_contested_owner = event_target:pfp_north_africa_winner
						remove_core_of = Q26
						remove_claim_by = Q26
						remove_contested_owner = Q26
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_north_africa_winner = { tag = Q27 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q27  has_contested_owner = Q27 } }
						add_core_of = event_target:pfp_north_africa_winner
						add_contested_owner = event_target:pfp_north_africa_winner
						remove_core_of = Q27
						remove_claim_by = Q27
						remove_contested_owner = Q27
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_north_africa_winner = { tag = Q28 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q28  has_contested_owner = Q28 } }
						add_core_of = event_target:pfp_north_africa_winner
						add_contested_owner = event_target:pfp_north_africa_winner
						remove_core_of = Q28
						remove_claim_by = Q28
						remove_contested_owner = Q28
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_north_africa_winner = { tag = Q29 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q29  has_contested_owner = Q29 } }
						add_core_of = event_target:pfp_north_africa_winner
						add_contested_owner = event_target:pfp_north_africa_winner
						remove_core_of = Q29
						remove_claim_by = Q29
						remove_contested_owner = Q29
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_north_africa_winner = { tag = Q30 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q30  has_contested_owner = Q30 } }
						add_core_of = event_target:pfp_north_africa_winner
						add_contested_owner = event_target:pfp_north_africa_winner
						remove_core_of = Q30
						remove_claim_by = Q30
						remove_contested_owner = Q30
					}
				}

				# Transfer pfp_possible_cores arrays, annex losers, set tag and flag
				event_target:pfp_north_africa_winner = {
					if = {
						limit = { NOT = { tag = Q26 } }
						for_each_loop = {
							array = Q26.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = {
						limit = { NOT = { tag = Q27 } }
						for_each_loop = {
							array = Q27.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = {
						limit = { NOT = { tag = Q28 } }
						for_each_loop = {
							array = Q28.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = {
						limit = { NOT = { tag = Q29 } }
						for_each_loop = {
							array = Q29.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = {
						limit = { NOT = { tag = Q30 } }
						for_each_loop = {
							array = Q30.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = { limit = { NOT = { tag = Q26 } } annex_country = { target = Q26 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q27 } } annex_country = { target = Q27 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q28 } } annex_country = { target = Q28 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q29 } } annex_country = { target = Q29 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q30 } } annex_country = { target = Q30 transfer_troops = yes } }
					set_cosmetic_tag = PFP_NORTH_AFRICA
					set_country_flag = pfp_is_north_africa
				}

				clear_global_event_target = pfp_north_africa_winner
				set_global_flag = pfp_north_africa_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed North Africa + Western Sahara (Q53) -> absorb Western Sahara
	# ----------------------------------------------------------------
	pfp_merge_north_africa_western_sahara = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_north_africa_merged
			NOT = { has_global_flag = pfp_north_africa_absorbed_western_sahara }
			Q53 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_north_africa_western_sahara_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_north_africa_absorbed_western_sahara }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_north_africa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_north_africa }
			}
			Q53 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_north_africa_western_sahara_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_north_africa }
					save_global_event_target_as = pfp_north_africa_country
					for_each_loop = {
						array = Q53.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = Q53  has_contested_owner = Q53 } }
					add_core_of = event_target:pfp_north_africa_country
					add_contested_owner = event_target:pfp_north_africa_country
					remove_core_of = Q53
					remove_claim_by = Q53
					remove_contested_owner = Q53
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_north_africa }
					annex_country = { target = Q53 transfer_troops = yes }
				}

				clear_global_event_target = pfp_north_africa_country
				set_global_flag = pfp_north_africa_absorbed_western_sahara
			}
		}

		ai_will_do = { factor = 0 }
	}

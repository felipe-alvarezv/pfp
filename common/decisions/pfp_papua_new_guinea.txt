pfp_category = {
	pfp_papua_new_guinea = {
		allowed = {
			NOT = {
				original_tag = PNG #Papua New Guinea
			}
		}
		available = {
			is_subject = no
			#Papua
			custom_trigger_tooltip = {
				tooltip = pfp_papua_new_guinea_required_states_papua
				has_full_control_of_state = 523 #Papua
			}
			#Interior Papua
			custom_trigger_tooltip = {
				tooltip = pfp_papua_new_guinea_required_states_interior_papua
				has_full_control_of_state = 1073 #Interior Papua
			}
			#New Guinea
			custom_trigger_tooltip = {
				tooltip = pfp_papua_new_guinea_required_states_new_guinea
				has_full_control_of_state = 979 #New Guinea
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 523 #Papua
				has_full_control_of_state = 1073 #Interior Papua
				has_full_control_of_state = 979 #New Guinea
			}
			NOT = { has_global_flag = pfp_papua_new_guinea_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 523 #Papua
				state = 1073 #Interior Papua
				state = 979 #New Guinea
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_papua_new_guinea_transfer_capital
			custom_effect_tooltip = pfp_papua_new_guinea_puppet
			custom_effect_tooltip = pfp_papua_new_guinea_occupation_type
			custom_effect_tooltip = pfp_papua_new_guinea_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				523 = { transfer_state_to = R21 } #Papua

				#Formable puppet becomes a puppet of ROOT
				puppet = R21

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R21 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_papua_new_guinea = yes

				#Transfer all technology from ROOT to formable puppet
				R21 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_papua_new_guinea_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

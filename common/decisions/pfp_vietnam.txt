pfp_category = {
	pfp_vietnam = {
		allowed = {
			NOT = {
				original_tag = VIN #Vietnam
			}
		}
		available = {
			is_subject = no
			#Tonkin
			custom_trigger_tooltip = {
				tooltip = pfp_vietnam_required_states_tonkin
				has_full_control_of_state = 671 #Tonkin
			}
			#Da Nang
			custom_trigger_tooltip = {
				tooltip = pfp_vietnam_required_states_da_nang
				has_full_control_of_state = 1017 #Da Nang
			}
			#South Annam
			custom_trigger_tooltip = {
				tooltip = pfp_vietnam_required_states_south_annam
				has_full_control_of_state = 286 #South Annam
			}
			#Cochinchina
			custom_trigger_tooltip = {
				tooltip = pfp_vietnam_required_states_cochinchina
				has_full_control_of_state = 1066 #Cochinchina
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 671 #Tonkin
				has_full_control_of_state = 1017 #Da Nang
				has_full_control_of_state = 286 #South Annam
				has_full_control_of_state = 1066 #Cochinchina
			}
			NOT = { has_global_flag = pfp_vietnam_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 671 #Tonkin
				state = 1017 #Da Nang
				state = 286 #South Annam
				state = 1066 #Cochinchina
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_vietnam_transfer_capital
			custom_effect_tooltip = pfp_vietnam_puppet
			custom_effect_tooltip = pfp_vietnam_occupation_type
			custom_effect_tooltip = pfp_vietnam_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				671 = { transfer_state_to = R12 } #Tonkin

				#Formable puppet becomes a puppet of ROOT
				puppet = R12

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R12 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_vietnam = yes

				#Transfer all technology from ROOT to formable puppet
				R12 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_vietnam_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

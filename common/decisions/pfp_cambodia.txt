pfp_category = {
	pfp_cambodia = {
		allowed = {
			NOT = {
				original_tag = CAM #Cambodia
			}
		}
		available = {
			is_subject = no
			#Cambodia
			custom_trigger_tooltip = {
				tooltip = pfp_cambodia_required_states_cambodia
				has_full_control_of_state = 741 #Cambodia
			}
			#Battambang
			custom_trigger_tooltip = {
				tooltip = pfp_cambodia_required_states_battambang
				has_full_control_of_state = 1067 #Battambang
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 741 #Cambodia
				has_full_control_of_state = 1067 #Battambang
			}
			NOT = { has_global_flag = pfp_cambodia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 741 #Cambodia
				state = 1067 #Battambang
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_cambodia_transfer_capital
			custom_effect_tooltip = pfp_cambodia_puppet
			custom_effect_tooltip = pfp_cambodia_occupation_type
			custom_effect_tooltip = pfp_cambodia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				741 = { transfer_state_to = R14 } #Cambodia

				#Formable puppet becomes a puppet of ROOT
				puppet = R14

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R14 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_cambodia = yes

				#Transfer all technology from ROOT to formable puppet
				R14 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_cambodia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

###########################################################################
# MERGE DECISIONS - CENTRAL AMERICA
# Allows the overlord to merge Central American puppet states into a
# unified Central America.
#
# Decision 1: pfp_merge_central_america
#   Requires Guatemala (Q90), Honduras (Q91), El Salvador (Q92),
#   Nicaragua (Q93), Costa Rica (Q94), and Panama (Q95) as PFP subjects.
#   The puppet with the most total factories becomes Central America.
#
# Follow-up decisions (available after Central America is formed):
#   pfp_merge_central_america_belize - Absorbs Belize (R08)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Guatemala (Q90) + Honduras (Q91) + El Salvador (Q92) +
	# Nicaragua (Q93) + Costa Rica (Q94) + Panama (Q95) -> Central America
	# ----------------------------------------------------------------
	pfp_merge_central_america = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_central_america_merged }
			OR = {
				Q90 = { is_subject_of = ROOT }
				Q91 = { is_subject_of = ROOT }
				Q92 = { is_subject_of = ROOT }
				Q93 = { is_subject_of = ROOT }
				Q94 = { is_subject_of = ROOT }
				Q95 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_america_not_yet_formed_tt
				NOT = { has_global_flag = pfp_central_america_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_america_puppets_ready_tt
				Q90 = { is_subject_of = ROOT }
				Q91 = { is_subject_of = ROOT }
				Q92 = { is_subject_of = ROOT }
				Q93 = { is_subject_of = ROOT }
				Q94 = { is_subject_of = ROOT }
				Q95 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				Q90 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q91 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q92 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q93 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q94 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q95 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				Q90 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q91 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q92 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q93 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q94 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q95 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_central_america_effect_form
			custom_effect_tooltip = pfp_merge_central_america_effect_annex
			custom_effect_tooltip = pfp_merge_central_america_effect_cores
			hidden_effect = {
				# Each subsequent puppet wins if it strictly has more factories.
				Q90 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_central_america_winner
				}
				Q91 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_central_america_winner.pfp_fac } }
						save_global_event_target_as = pfp_central_america_winner
					}
				}
				Q92 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_central_america_winner.pfp_fac } }
						save_global_event_target_as = pfp_central_america_winner
					}
				}
				Q93 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_central_america_winner.pfp_fac } }
						save_global_event_target_as = pfp_central_america_winner
					}
				}
				Q94 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_central_america_winner.pfp_fac } }
						save_global_event_target_as = pfp_central_america_winner
					}
				}
				Q95 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_central_america_winner.pfp_fac } }
						save_global_event_target_as = pfp_central_america_winner
					}
				}

				# Transfer state cores/ownership from losers to winner
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q90 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q90  has_contested_owner = Q90 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q90
						remove_claim_by = Q90
						remove_contested_owner = Q90
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q91 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q91  has_contested_owner = Q91 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q91
						remove_claim_by = Q91
						remove_contested_owner = Q91
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q92 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q92  has_contested_owner = Q92 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q92
						remove_claim_by = Q92
						remove_contested_owner = Q92
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q93 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q93  has_contested_owner = Q93 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q93
						remove_claim_by = Q93
						remove_contested_owner = Q93
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q94 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q94  has_contested_owner = Q94 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q94
						remove_claim_by = Q94
						remove_contested_owner = Q94
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_central_america_winner = { tag = Q95 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q95  has_contested_owner = Q95 } }
						add_core_of = event_target:pfp_central_america_winner
						add_contested_owner = event_target:pfp_central_america_winner
						remove_core_of = Q95
						remove_claim_by = Q95
						remove_contested_owner = Q95
					}
				}

				# Transfer pfp_possible_cores arrays, annex losers, set tag and flag
				event_target:pfp_central_america_winner = {
					if = { limit = { NOT = { tag = Q90 } } for_each_loop = { array = Q90.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q91 } } for_each_loop = { array = Q91.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q92 } } for_each_loop = { array = Q92.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q93 } } for_each_loop = { array = Q93.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q94 } } for_each_loop = { array = Q94.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q95 } } for_each_loop = { array = Q95.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q90 } } annex_country = { target = Q90 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q91 } } annex_country = { target = Q91 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q92 } } annex_country = { target = Q92 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q93 } } annex_country = { target = Q93 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q94 } } annex_country = { target = Q94 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q95 } } annex_country = { target = Q95 transfer_troops = yes } }
					set_cosmetic_tag = PFP_CENTRAL_AMERICA
					set_country_flag = pfp_is_central_america
				}

				clear_global_event_target = pfp_central_america_winner
				set_global_flag = pfp_central_america_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Central America + Belize (R08) -> absorb Belize
	# ----------------------------------------------------------------
	pfp_merge_central_america_belize = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_central_america_merged
			NOT = { has_global_flag = pfp_central_america_absorbed_belize }
			R08 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_america_belize_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_central_america_absorbed_belize }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_america_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_central_america }
			}
			R08 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
					has_autonomy_state = pfp_autonomy_military_occupation
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_central_america_belize_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_central_america }
					save_global_event_target_as = pfp_central_america_country
					for_each_loop = {
						array = R08.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = R08  has_contested_owner = R08 } }
					add_core_of = event_target:pfp_central_america_country
					add_contested_owner = event_target:pfp_central_america_country
					remove_core_of = R08
					remove_claim_by = R08
					remove_contested_owner = R08
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_central_america }
					annex_country = { target = R08 transfer_troops = yes }
				}
				clear_global_event_target = pfp_central_america_country
				set_global_flag = pfp_central_america_absorbed_belize
			}
		}

		ai_will_do = { factor = 0 }
	}

}

pfp_category = {
	pfp_syria = {
		allowed = {
			NOT = {
				original_tag = SYR #Syria
			}
		}
		available = {
			is_subject = no
			#Damascus
			custom_trigger_tooltip = {
				tooltip = pfp_syria_required_states_damascus
				has_full_control_of_state = 554 #Damascus
			}
			#Deir-az-Zur
			custom_trigger_tooltip = {
				tooltip = pfp_syria_required_states_deir-az-zur
				has_full_control_of_state = 680 #Deir-az-Zur
			}
			#Aleppo
			custom_trigger_tooltip = {
				tooltip = pfp_syria_required_states_aleppo
				has_full_control_of_state = 677 #Aleppo
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 554 #Damascus
				has_full_control_of_state = 680 #Deir-az-Zur
				has_full_control_of_state = 677 #Aleppo
			}
			NOT = { has_global_flag = pfp_syria_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 554 #Damascus
				state = 680 #Deir-az-Zur
				state = 677 #Aleppo
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_syria_transfer_capital
			custom_effect_tooltip = pfp_syria_puppet
			custom_effect_tooltip = pfp_syria_occupation_type
			custom_effect_tooltip = pfp_syria_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				554 = { transfer_state_to = Q89 } #Damascus

				#Formable puppet becomes a puppet of ROOT
				puppet = Q89

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q89 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_syria = yes

				#Transfer all technology from ROOT to formable puppet
				Q89 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_syria_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

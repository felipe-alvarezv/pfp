###########################################################################
# MERGE DECISIONS - EAST AFRICA
# Allows the overlord to merge East African puppet states into a unified
# East Africa.
#
# Decision 1: pfp_merge_east_africa
#   Requires all 5 Horn of Africa puppets as PFP subjects:
#   Ethiopia (Q55), Eritrea (Q56), Djibouti (Q57), Somaliland (Q58),
#   Somalia (Q59).
#   The puppet with the most total factories + scaled manpower becomes
#   East Africa.
#
# Follow-up decisions (available after East Africa is formed, all independent):
#   pfp_merge_east_africa_kenya        - Absorbs Kenya (Q60)
#   pfp_merge_east_africa_uganda       - Absorbs Uganda (Q61)
#   pfp_merge_east_africa_tanganyika   - Absorbs Tanganyika (Q62)
#   pfp_merge_east_africa_ruanda_urundi - Absorbs Ruanda-Urundi (Q66)
###########################################################################

						has_autonomy_state = pfp_autonomy_military_occupation_painted

						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}

				Q56 = {
					is_subject_of = ROOT

						has_autonomy_state = pfp_autonomy_military_occupation

				set_global_flag = pfp_east_africa_merged

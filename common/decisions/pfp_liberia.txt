pfp_category = {
	pfp_liberia = {
		allowed = {
			NOT = {
				original_tag = LIB #Liberia
			}
		}
		available = {
			is_subject = no
			#Liberia
			custom_trigger_tooltip = {
				tooltip = pfp_liberia_required_states_liberia
				has_full_control_of_state = 298 #Liberia
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 298 #Liberia
			}
			NOT = { has_global_flag = pfp_liberia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 298 #Liberia
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_liberia_transfer_capital
			custom_effect_tooltip = pfp_liberia_puppet
			custom_effect_tooltip = pfp_liberia_occupation_type
			custom_effect_tooltip = pfp_liberia_state_controller

			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet

				puppet = Q45

				#Code to assign generic portraits and set puppet color

				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes

						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_liberia = yes

				#Transfer all technology from ROOT to formable puppet
				Q45 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_liberia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

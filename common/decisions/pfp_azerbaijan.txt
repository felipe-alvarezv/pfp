pfp_category = {
	pfp_azerbaijan = {
		allowed = {
			NOT = {
				original_tag = AZR #Azerbaijan
			}
		}
		available = {
			is_subject = no
			#Azerbaijan (229)
			custom_trigger_tooltip = {
				tooltip = pfp_azerbaijan_required_states_azerbaijan
				has_full_control_of_state = 229 #Azerbaijan
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 229 #Azerbaijan
			}
			NOT = { has_global_flag = pfp_azerbaijan_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 229 #Azerbaijan
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_azerbaijan_transfer_capital
			custom_effect_tooltip = pfp_azerbaijan_puppet
			custom_effect_tooltip = pfp_azerbaijan_occupation_type
			custom_effect_tooltip = pfp_azerbaijan_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				229 = { transfer_state_to = R29 } #Azerbaijan

				#Formable puppet becomes a puppet of ROOT
				puppet = R29

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R29 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_azerbaijan = yes

				#Transfer all technology from ROOT to formable puppet
				R29 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_azerbaijan_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

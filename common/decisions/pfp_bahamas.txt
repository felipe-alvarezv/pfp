pfp_category = {
	pfp_bahamas = {
		allowed = {
			NOT = {
				original_tag = BAH #Bahamas
			}
		}
		available = {
			is_subject = no
			#Bahamas
			custom_trigger_tooltip = {
				tooltip = pfp_bahamas_required_states_northern_bahamas
				has_full_control_of_state = 690 #Northern Bahamas
			}
			custom_trigger_tooltip = {
				tooltip = pfp_bahamas_required_states_southern_bahamas
				has_full_control_of_state = 693 #Southern Bahamas
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 690 #Northern Bahamas
				has_full_control_of_state = 693 #Southern Bahamas
			}
			NOT = { has_global_flag = pfp_bahamas_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 690 #Northern Bahamas
				state = 693 #Southern Bahamas
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_bahamas_transfer_capital
			custom_effect_tooltip = pfp_bahamas_puppet
			custom_effect_tooltip = pfp_bahamas_occupation_type
			custom_effect_tooltip = pfp_bahamas_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				690 = { transfer_state_to = R01 } #Northern Bahamas

				#Formable puppet becomes a puppet of ROOT
				puppet = R01

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R01 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_bahamas = yes

				#Transfer all technology from ROOT to formable puppet
				R01 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_bahamas_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

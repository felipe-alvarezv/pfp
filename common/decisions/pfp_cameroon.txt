	pfp_cameroon = {
		allowed = {
			NOT = {
				original_tag = CMR #Cameroon
			}
		}
		available = {
			is_subject = no
			#Cameroon
			custom_trigger_tooltip = {
				tooltip = pfp_cameroon_required_states_cameroon
				has_full_control_of_state = 773 #Cameroon
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 773 #Cameroon
			}
			NOT = { has_global_flag = pfp_cameroon_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 773 #Cameroon
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_cameroon_transfer_capital
			custom_effect_tooltip = pfp_cameroon_puppet
			custom_effect_tooltip = pfp_cameroon_occupation_type

						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}

					}
				}

				Q36 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_cameroon_flag
			}
		}
		ai_will_do = { factor = 0 }
	}

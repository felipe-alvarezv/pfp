pfp_category = {
	pfp_georgia = {
		allowed = {
			NOT = {
				original_tag = GEO #Georgia
			}
		}
		available = {
			is_subject = no
			#Georgia (231)
			custom_trigger_tooltip = {
				tooltip = pfp_georgia_required_states_georgia
				has_full_control_of_state = 231 #Georgia
			}
			#Abkhazia (826)
			custom_trigger_tooltip = {
				tooltip = pfp_georgia_required_states_abkhazia
				has_full_control_of_state = 826 #Abkhazia
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 231 #Georgia
				has_full_control_of_state = 826 #Abkhazia
			}
			NOT = { has_global_flag = pfp_georgia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 231 #Georgia
				state = 826 #Abkhazia
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_georgia_transfer_capital
			custom_effect_tooltip = pfp_georgia_puppet
			custom_effect_tooltip = pfp_georgia_occupation_type
			custom_effect_tooltip = pfp_georgia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				231 = { transfer_state_to = R27 } #Georgia

				#Formable puppet becomes a puppet of ROOT
				puppet = R27

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R27 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_georgia = yes

				#Transfer all technology from ROOT to formable puppet
				R27 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_georgia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

pfp_category = {
	pfp_new_caledonia = {
		allowed = {
			NOT = {
				original_tag = TAH #New Caledonia
			}
		}
		available = {
			is_subject = no
			#New Caledonia
			custom_trigger_tooltip = {
				tooltip = pfp_new_caledonia_required_states_new_caledonia
				has_full_control_of_state = 635 #New Caledonia
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 635 #New Caledonia
			}
			NOT = { has_global_flag = pfp_new_caledonia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 635 #New Caledonia
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_new_caledonia_transfer_capital
			custom_effect_tooltip = pfp_new_caledonia_puppet
			custom_effect_tooltip = pfp_new_caledonia_occupation_type
			custom_effect_tooltip = pfp_new_caledonia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				635 = { transfer_state_to = R24 } #New Caledonia

				#Formable puppet becomes a puppet of ROOT
				puppet = R24

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R24 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = { has_autonomy_state = pfp_autonomy_military_occupation }
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_new_caledonia = yes

				#Transfer all technology from ROOT to formable puppet
				R24 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_new_caledonia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

pfp_category = {
	pfp_ubangi_shari = {
		allowed = {
			NOT = {
				original_tag = CAR #Central African Republic
			}
		}
		available = {
			is_subject = no
			#Equatorial Africa
			custom_trigger_tooltip = {
				tooltip = pfp_ubangi_shari_required_states_equatorial_africa
				has_full_control_of_state = 660 #Equatorial Africa
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 660 #Equatorial Africa
			}
			NOT = { has_global_flag = pfp_ubangi_shari_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 660 #Equatorial Africa
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_ubangi_shari_transfer_capital
			custom_effect_tooltip = pfp_ubangi_shari_puppet
			custom_effect_tooltip = pfp_ubangi_shari_occupation_type
			custom_effect_tooltip = pfp_ubangi_shari_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				660 = { transfer_state_to = Q37 } #Equatorial Africa

				#Formable puppet becomes a puppet of ROOT
				puppet = Q37

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q37 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_ubangi_shari = yes

				#Transfer all technology from ROOT to formable puppet
				Q37 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_ubangi_shari_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

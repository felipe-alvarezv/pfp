###########################################################################
# MERGE DECISIONS - GREAT BRITAIN
# Allows the overlord to merge British puppet states into a unified nation.
#
# Decision 1: pfp_merge_great_britain
#   Requires England (P80), Scotland (P81), and Wales (P82) as PFP subjects.
#   The puppet with the most total factories + scaled manpower becomes
#   Great Britain (PFP_GREAT_BRITAIN).
#
# Follow-up decision:
#   pfp_merge_great_britain_northern_ireland - Absorbs Northern Ireland (P84)
#   and renames the country to Great Britain and Northern Ireland (PFP_UK).
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# England + Scotland + Wales -> Great Britain
	# ----------------------------------------------------------------
	pfp_merge_great_britain = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_great_britain_merged }
			OR = {
				P80 = { is_subject_of = ROOT }
				P81 = { is_subject_of = ROOT }
				P82 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_great_britain_not_yet_formed_tt
				NOT = { has_global_flag = pfp_great_britain_merged }
			}
			P80 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			P81 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			P82 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			NOT = { P80 = { any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } } } }
			NOT = { P81 = { any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } } } }
			NOT = { P82 = { any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } } } }
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_great_britain_effect_form
			custom_effect_tooltip = pfp_merge_great_britain_effect_annex
			custom_effect_tooltip = pfp_merge_great_britain_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories + (manpower / 100000)
				P80 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P81 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P82 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }

				# Initialize winner, then linear challenge
				P80 = { save_global_event_target_as = pfp_great_britain_winner }
				if = { limit = { check_variable = { P81.pfp_fac > event_target:pfp_great_britain_winner.pfp_fac } } P81 = { save_global_event_target_as = pfp_great_britain_winner } }
				if = { limit = { check_variable = { P82.pfp_fac > event_target:pfp_great_britain_winner.pfp_fac } } P82 = { save_global_event_target_as = pfp_great_britain_winner } }

				# Transfer state cores/ownership from non-winner subjects to winner
				if = {
					limit = { NOT = { event_target:pfp_great_britain_winner = { tag = P80 } } }
					every_state = {
						limit = { OR = { is_owned_by = P80  has_contested_owner = P80 } }
						add_core_of = event_target:pfp_great_britain_winner
						add_contested_owner = event_target:pfp_great_britain_winner
						remove_core_of = P80
						remove_claim_by = P80
						remove_contested_owner = P80
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_great_britain_winner = { tag = P81 } } }
					every_state = {
						limit = { OR = { is_owned_by = P81  has_contested_owner = P81 } }
						add_core_of = event_target:pfp_great_britain_winner
						add_contested_owner = event_target:pfp_great_britain_winner
						remove_core_of = P81
						remove_claim_by = P81
						remove_contested_owner = P81
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_great_britain_winner = { tag = P82 } } }
					every_state = {
						limit = { OR = { is_owned_by = P82  has_contested_owner = P82 } }
						add_core_of = event_target:pfp_great_britain_winner
						add_contested_owner = event_target:pfp_great_britain_winner
						remove_core_of = P82
						remove_claim_by = P82
						remove_contested_owner = P82
					}
				}

				# Transfer pfp_possible_cores arrays, annex non-winners, set tag and flags
				event_target:pfp_great_britain_winner = {
					if = { limit = { NOT = { tag = P80 } } for_each_loop = { array = P80.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P81 } } for_each_loop = { array = P81.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P82 } } for_each_loop = { array = P82.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P80 } } annex_country = { target = P80 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P81 } } annex_country = { target = P81 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P82 } } annex_country = { target = P82 transfer_troops = yes } }
					set_cosmetic_tag = PFP_GREAT_BRITAIN
					set_country_flag = pfp_is_great_britain
				}

				clear_global_event_target = pfp_great_britain_winner
				set_global_flag = pfp_great_britain_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Great Britain + Northern Ireland (P84) -> Great Britain
	# and Northern Ireland (PFP_UK cosmetic tag)
	# ----------------------------------------------------------------
	pfp_merge_great_britain_northern_ireland = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_great_britain_merged
			NOT = { has_global_flag = pfp_great_britain_absorbed_northern_ireland }
			P84 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_great_britain_northern_ireland_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_great_britain_absorbed_northern_ireland }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_great_britain_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_great_britain }
			}
			P84 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_great_britain_northern_ireland_effect_annex
			custom_effect_tooltip = pfp_merge_great_britain_northern_ireland_effect_uk
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_great_britain }
					save_global_event_target_as = pfp_great_britain_country
					for_each_loop = {
						array = P84.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = P84  has_contested_owner = P84 } }
					add_core_of = event_target:pfp_great_britain_country
					add_contested_owner = event_target:pfp_great_britain_country
					remove_core_of = P84
					remove_claim_by = P84
					remove_contested_owner = P84
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_great_britain }
					annex_country = { target = P84 transfer_troops = yes }
					set_cosmetic_tag = PFP_UK
				}

				clear_global_event_target = pfp_great_britain_country
				set_global_flag = pfp_great_britain_absorbed_northern_ireland
			}
		}

		ai_will_do = { factor = 0 }
	}

}

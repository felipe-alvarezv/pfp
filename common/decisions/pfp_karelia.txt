pfp_category = {
	pfp_karelia = {
		allowed = {
			NOT = {
				original_tag = KAR #Karelia
			}
		}
		available = {
			is_subject = no
			#Karelia
			custom_trigger_tooltip = {
				tooltip = pfp_karelia_required_states_onega
				has_full_control_of_state = 215 #Onega
			}
			custom_trigger_tooltip = {
				tooltip = pfp_karelia_required_states_olonets
				has_full_control_of_state = 216 #Olonets
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 215 #Onega
				has_full_control_of_state = 216 #Olonets
			}
			NOT = { has_global_flag = pfp_karelia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 215 #Onega
				state = 216 #Olonets
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_karelia_transfer_capital
			custom_effect_tooltip = pfp_karelia_puppet
			custom_effect_tooltip = pfp_karelia_occupation_type
			custom_effect_tooltip = pfp_karelia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				215 = { transfer_state_to = Q76 } #Onega

				#Formable puppet becomes a puppet of ROOT
				puppet = Q76

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q76 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_karelia = yes

				#Transfer all technology from ROOT to formable puppet
				Q76 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_karelia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

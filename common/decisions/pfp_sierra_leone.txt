	pfp_sierra_leone = {
		allowed = {
			NOT = {
				original_tag = SIE #Sierra Leone
			}
		}
		available = {
			is_subject = no

			custom_trigger_tooltip = {
				tooltip = pfp_sierra_leone_required_states_sierra_leone
				has_full_control_of_state = 700 #Sierra Leone
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 700 #Sierra Leone
			}
			NOT = { has_global_flag = pfp_sierra_leone_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 700 #Sierra Leone
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_sierra_leone_transfer_capital

								has_autonomy_state = pfp_autonomy_military_occupation_painted

				set_global_flag = pfp_sierra_leone_flag
			}
		}
		ai_will_do = { factor = 0 }

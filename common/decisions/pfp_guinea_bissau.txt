pfp_category = {
	pfp_guinea_bissau = {
		allowed = {
			NOT = {
				original_tag = GNB #Guinea-Bissau
			}
		}
		available = {
			is_subject = no
			#Guinea-Bissau
			custom_trigger_tooltip = {
				tooltip = pfp_guinea_bissau_required_states_guinea_bissau
				has_full_control_of_state = 296 #Portuguese Guinea
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 296 #Portuguese Guinea
			}
			NOT = { has_global_flag = pfp_guinea_bissau_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 296 #Portuguese Guinea
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_guinea_bissau_transfer_capital
			custom_effect_tooltip = pfp_guinea_bissau_puppet
			custom_effect_tooltip = pfp_guinea_bissau_occupation_type
			custom_effect_tooltip = pfp_guinea_bissau_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				296 = { transfer_state_to = Q48 } #Portuguese Guinea

				#Formable puppet becomes a puppet of ROOT
				puppet = Q48

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q48 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_guinea_bissau = yes

				#Transfer all technology from ROOT to formable puppet
				Q48 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_guinea_bissau_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

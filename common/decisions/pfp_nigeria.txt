		available = {

				tooltip = pfp_nigeria_required_states_lagos
				has_full_control_of_state = 558 #Lagos
			}

			custom_trigger_tooltip = {
				tooltip = pfp_nigeria_required_states_sokoto

			custom_trigger_tooltip = {
				tooltip = pfp_nigeria_required_states_borno

				tooltip = pfp_nigeria_required_states_benue
				has_full_control_of_state = 900 #Benue
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 558 #Lagos

								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted

				set_global_flag = pfp_nigeria_flag
			}

		ai_will_do = { factor = 0 }

###########################################################################
# MERGE DECISIONS - CENTRAL AFRICA
# Allows the overlord to merge Central African puppet states into a unified
# Central Africa.
#
# Decision 1: pfp_merge_central_africa
#   Requires the Belgian Congo (Q65).
#   The Belgian Congo becomes Central Africa.
#
# Follow-up decisions (available after Central Africa is formed):
#   pfp_merge_central_africa_congo_brazzaville - Absorbs Congo-Brazzaville (Q39)
#   pfp_merge_central_africa_gabon             - Absorbs Gabon (Q38)
#   pfp_merge_central_africa_ubangi_shari      - Absorbs Ubangi-Shari (Q37)
#   pfp_merge_central_africa_cameroon          - Absorbs Cameroon (Q36)
#   pfp_merge_central_africa_chad              - Absorbs Chad (Q32)
#   pfp_merge_central_africa_equatorial_guinea - Absorbs Equatorial Guinea (Q54)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Belgian Congo (Q65) -> Central Africa
	# ----------------------------------------------------------------
	pfp_merge_central_africa = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_central_africa_merged }
			Q65 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_africa_not_yet_formed_tt
				NOT = { has_global_flag = pfp_central_africa_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_africa_puppets_ready_tt
				Q65 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_central_africa_effect_form
			custom_effect_tooltip = pfp_merge_central_africa_effect_annex
			custom_effect_tooltip = pfp_merge_central_africa_effect_cores
			hidden_effect = {
				# Q65 Belgian Congo is the sole territory — initialize as winner directly.
				Q65 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_central_africa_winner
				}

				event_target:pfp_central_africa_winner = {
					set_cosmetic_tag = PFP_CENTRAL_AFRICA
					set_country_flag = pfp_is_central_africa
				}

				clear_global_event_target = pfp_central_africa_winner
				set_global_flag = pfp_central_africa_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Central Africa + Congo-Brazzaville (Q39)
	# ----------------------------------------------------------------
	pfp_merge_central_africa_congo_brazzaville = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_central_africa_merged
			NOT = { has_global_flag = pfp_central_africa_absorbed_congo_brazzaville }
			Q39 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_africa_congo_brazzaville_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_central_africa_absorbed_congo_brazzaville }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_central_africa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_central_africa }
			}
			Q39 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_central_africa_congo_brazzaville_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_central_africa }
					save_global_event_target_as = pfp_central_africa_country
					for_each_loop = {
						array = Q39.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = Q39  has_contested_owner = Q39 } }
					add_core_of = event_target:pfp_central_africa_country
					add_contested_owner = event_target:pfp_central_africa_country
					remove_core_of = Q39
					remove_claim_by = Q39
					remove_contested_owner = Q39
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_central_africa }
					annex_country = { target = Q39 transfer_troops = yes }
				}
				clear_global_event_target = pfp_central_africa_country
				set_global_flag = pfp_central_africa_absorbed_congo_brazzaville
			}
		}

		ai_will_do = { factor = 0 }
	}

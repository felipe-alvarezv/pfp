pfp_puppet_management = {
	#Decision to show PFP GUI
    pfp_management_window = {
		icon = generic_industry

		available = { 
            is_ai = no
        }

		fire_only_once = no
		complete_effect = { 
            set_country_flag = pfp_management_window
        }
		ai_will_do = { factor = 0 }
	}

	#Decision to display overlord flag for puppets
	pfp_management_use_overlord_flag = {
		icon = generic_industry

		available = {
			is_ai = no
		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				if = {
					limit = { has_global_flag = pfp_use_overlord_flag }

					clr_global_flag = pfp_use_overlord_flag

					
					every_country = {
						every_subject_country  = {
							limit = { has_autonomy_state = pfp_autonomy_military_occupation }
							
							pfp_set_use_overlord_flag = yes
						}
					}
				}
				else = {
					set_global_flag = pfp_use_overlord_flag

					every_country = {
						every_subject_country  = {
							limit = { has_autonomy_state = pfp_autonomy_military_occupation }
							
							pfp_set_use_overlord_flag = yes
						}
					}
				}
			}
		}
	}

	#Decision to toggle resistance mechanics
	pfp_management_resistance = {
		icon = guerilla_base

		available = {
			is_ai = no
		}

		fire_only_once = no

		complete_effect = {
			hidden_effect = {
				if = {
					limit = { has_global_flag = pfp_management_resistance_enabled }

					clr_global_flag = pfp_management_resistance_enabled
				}
				else = {
					set_global_flag = pfp_management_resistance_enabled
				}
			}
		}
		ai_will_do = { factor = 0 }
	}
}

pfp_puppet_integration = {
	#Decision for formable puppet AI to core states with compliance > 80 and resistance < 20
	pfp_puppet_integration_core_state = {
		available = {
			custom_trigger_tooltip = {
				tooltip = pfp_puppet_integration_core_state_trigger_tooltip
				any_state_in = {
					array = ROOT.pfp_possible_cores
					is_fully_controlled_by = ROOT
					NOT = { is_core_of = ROOT }

					if = {
						limit = { has_global_flag = pfp_management_resistance_enabled }
						compliance > 80
						resistance < 20
					}
				}
			}
		}

		visible = {
			has_autonomy_state = pfp_autonomy_military_occupation
		}
		
		fire_only_once = no

		complete_effect = {
			custom_effect_tooltip  = pfp_puppet_integration_core_state_effect_tooltip
			hidden_effect  = {
				for_each_scope_loop = {
					array = ROOT.pfp_possible_cores

					if = {
						limit = {
							is_fully_controlled_by = ROOT
							NOT = { is_core_of = ROOT }

							if = {
								limit = { has_global_flag = pfp_management_resistance_enabled }
								compliance > 80
								resistance < 20
							}
						}
						add_core_of = ROOT
					}
				}
			}
		}
		ai_will_do = { factor = 1 }
	}
}

pfp_puppet_assistance = {
	#Decision to send manpower to formable puppets
	pfp_deploy_reinforcements = {
		icon = generic_industry
		target_array = subjects
		target_trigger = {
			FROM = {
				exists = yes
				has_autonomy_state = pfp_autonomy_military_occupation
			}
		}
		visible = {
			any_subject_country = {
				has_autonomy_state = pfp_autonomy_military_occupation
			}
		}
		available = {
			FROM = { exists = yes }
			ROOT = { has_manpower > 2000 }
		}
		days_remove = 30
		complete_effect = {
			ROOT = {
				add_manpower = -2000
			}
		}
		remove_effect = {
			FROM = {
				add_manpower = 2000
			}
		}
		ai_will_do = { factor = 0 }
	}

	#Decision to send infantry equipment to formable puppets
	pfp_ship_arms_to_garrison = {
		icon = generic_industry
		target_array = subjects
		target_trigger = {
			FROM = {
				exists = yes
				has_autonomy_state = pfp_autonomy_military_occupation
			}
		}
		visible = {
			any_subject_country = {
				has_autonomy_state = pfp_autonomy_military_occupation
			}
		}
		available = {
			FROM = { exists = yes }
			ROOT = { has_equipment = { infantry_equipment > 2000 } }
		}
		days_remove = 15
		complete_effect = {
			ROOT = {
				add_equipment_to_stockpile = {
					type = infantry_equipment
					amount = -2000
				}
			}
		}
		remove_effect = {
			FROM = {
				add_equipment_to_stockpile = {
					type = infantry_equipment
					amount = 2000
					producer = ROOT
				}
			}
		}
		ai_will_do = { factor = 0 }
	}
}

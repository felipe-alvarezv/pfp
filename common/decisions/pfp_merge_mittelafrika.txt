###########################################################################
# MERGE DECISIONS - MITTELAFRIKA
# Allows the overlord to merge the former German African colonies into a
# unified Mittelafrika, then absorb additional African puppets.
#
# Decision 1: pfp_merge_mittelafrika
#   Requires Cameroon (Q36), Tanganyika (Q62), and South West Africa (Q71).
#   The puppet with the most total factories becomes Mittelafrika.
#
# Follow-up decisions (available after Mittelafrika is formed):
#   pfp_merge_mittelafrika_togo              - Absorbs Togo (Q40)
#   pfp_merge_mittelafrika_ruanda_urundi     - Absorbs Ruanda-Urundi (Q66)
#   pfp_merge_mittelafrika_gabon             - Absorbs Gabon (Q38)
#   pfp_merge_mittelafrika_congo_brazzaville - Absorbs Congo-Brazzaville (Q39)
#   pfp_merge_mittelafrika_northern_rhodesia - Absorbs Northern Rhodesia (Q63)
#   pfp_merge_mittelafrika_southern_rhodesia - Absorbs Southern Rhodesia (Q69)
#   pfp_merge_mittelafrika_angola            - Absorbs Angola (Q67)
#   pfp_merge_mittelafrika_mozambique        - Absorbs Mozambique (Q68)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Cameroon (Q36) + Tanganyika (Q62) + South West Africa (Q71) -> Mittelafrika
	# ----------------------------------------------------------------
	pfp_merge_mittelafrika = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			original_tag = GER
			NOT = { has_global_flag = pfp_mittelafrika_merged }
			OR = {
				Q36 = { is_subject_of = ROOT }
				Q62 = { is_subject_of = ROOT }
				Q71 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_mittelafrika_not_yet_formed_tt
				NOT = { has_global_flag = pfp_mittelafrika_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_mittelafrika_puppets_ready_tt
				Q36 = { is_subject_of = ROOT }
				Q62 = { is_subject_of = ROOT }
				Q71 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				Q36 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q62 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q71 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				Q36 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q62 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q71 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_mittelafrika_effect_form
			custom_effect_tooltip = pfp_merge_mittelafrika_effect_annex
			custom_effect_tooltip = pfp_merge_mittelafrika_effect_cores
			hidden_effect = {
				Q36 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_mittelafrika_winner
				}
				Q62 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_mittelafrika_winner.pfp_fac } }
						save_global_event_target_as = pfp_mittelafrika_winner
					}
				}
				Q71 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_mittelafrika_winner.pfp_fac } }
						save_global_event_target_as = pfp_mittelafrika_winner
					}
				}

				if = {
					limit = { NOT = { event_target:pfp_mittelafrika_winner = { tag = Q36 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q36  has_contested_owner = Q36 } }
						add_core_of = event_target:pfp_mittelafrika_winner
						add_contested_owner = event_target:pfp_mittelafrika_winner
						remove_core_of = Q36
						remove_claim_by = Q36
						remove_contested_owner = Q36
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_mittelafrika_winner = { tag = Q62 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q62  has_contested_owner = Q62 } }
						add_core_of = event_target:pfp_mittelafrika_winner
						add_contested_owner = event_target:pfp_mittelafrika_winner
						remove_core_of = Q62
						remove_claim_by = Q62
						remove_contested_owner = Q62
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_mittelafrika_winner = { tag = Q71 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q71  has_contested_owner = Q71 } }
						add_core_of = event_target:pfp_mittelafrika_winner
						add_contested_owner = event_target:pfp_mittelafrika_winner
						remove_core_of = Q71
						remove_claim_by = Q71
						remove_contested_owner = Q71
					}
				}

				event_target:pfp_mittelafrika_winner = {
					if = { limit = { NOT = { tag = Q36 } } for_each_loop = { array = Q36.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q62 } } for_each_loop = { array = Q62.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q71 } } for_each_loop = { array = Q71.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q36 } } annex_country = { target = Q36 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q62 } } annex_country = { target = Q62 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q71 } } annex_country = { target = Q71 transfer_troops = yes } }
					set_cosmetic_tag = PFP_MITTELAFRIKA
					set_country_flag = pfp_is_mittelafrika
				}

				clear_global_event_target = pfp_mittelafrika_winner
				set_global_flag = pfp_mittelafrika_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

pfp_category = {
	pfp_ruanda_urundi = {
		allowed = {
			NOT = {
				original_tag = RWA #Rwanda
			}
			NOT = {
				original_tag = BRD #Burundi
			}
		}
		available = {
			is_subject = no
			#Ruanda-Urundi
			custom_trigger_tooltip = {
				tooltip = pfp_ruanda_urundi_required_states_rwanda
				has_full_control_of_state = 768 #Rwanda
			}
			custom_trigger_tooltip = {
				tooltip = pfp_ruanda_urundi_required_states_burundi
				has_full_control_of_state = 769 #Burundi
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 768 #Rwanda
				has_full_control_of_state = 769 #Burundi
			}
			NOT = { has_global_flag = pfp_ruanda_urundi_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 768 #Rwanda
				state = 769 #Burundi
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_ruanda_urundi_transfer_capital
			custom_effect_tooltip = pfp_ruanda_urundi_puppet
			custom_effect_tooltip = pfp_ruanda_urundi_occupation_type
			custom_effect_tooltip = pfp_ruanda_urundi_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				769 = { transfer_state_to = Q66 } #Burundi

				#Formable puppet becomes a puppet of ROOT
				puppet = Q66

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q66 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_ruanda_urundi = yes

				#Transfer all technology from ROOT to formable puppet
				Q66 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_ruanda_urundi_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

###########################################################################
# MERGE DECISIONS - UNITED STATES
# Three-step merge system:
#   1. pfp_merge_western_usa: P21+P23 -> PFP_WESTERN_USA
#   2. pfp_merge_usa: P18+P20+P26 -> PFP_EASTERN_USA
#   3. pfp_merge_united_states: PFP_WESTERN_USA+PFP_EASTERN_USA -> PFP_USA
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Pacific U.S. (P21) + Rockies U.S. (P23) -> Western United States
	# ----------------------------------------------------------------
	pfp_merge_western_usa = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_western_usa_merged }
			OR = {
				P21 = { is_subject_of = ROOT }
				P23 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_western_usa_not_yet_formed_tt
				NOT = { has_global_flag = pfp_western_usa_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_western_usa_puppets_ready_tt
				P21 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P23 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_western_usa_effect_form
			custom_effect_tooltip = pfp_merge_western_usa_effect_annex
			custom_effect_tooltip = pfp_merge_western_usa_effect_cores
			hidden_effect = {
				P21 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }
				P23 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } }

				P21 = { save_global_event_target_as = pfp_western_usa_winner }
				if = { limit = { check_variable = { P23.pfp_fac > event_target:pfp_western_usa_winner.pfp_fac } } P23 = { save_global_event_target_as = pfp_western_usa_winner } }

				if = {
					limit = { NOT = { event_target:pfp_western_usa_winner = { tag = P21 } } }
					every_state = { limit = { OR = { is_owned_by = P21 has_contested_owner = P21 } } add_core_of = event_target:pfp_western_usa_winner add_contested_owner = event_target:pfp_western_usa_winner remove_core_of = P21 remove_claim_by = P21 remove_contested_owner = P21 }
				}
				if = {
					limit = { NOT = { event_target:pfp_western_usa_winner = { tag = P23 } } }
					every_state = { limit = { OR = { is_owned_by = P23 has_contested_owner = P23 } } add_core_of = event_target:pfp_western_usa_winner add_contested_owner = event_target:pfp_western_usa_winner remove_core_of = P23 remove_claim_by = P23 remove_contested_owner = P23 }
				}

				event_target:pfp_western_usa_winner = {
					if = { limit = { NOT = { tag = P21 } } for_each_loop = { array = P21.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P23 } } for_each_loop = { array = P23.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P21 } } annex_country = { target = P21 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P23 } } annex_country = { target = P23 transfer_troops = yes } }
					set_cosmetic_tag = PFP_WESTERN_USA
					set_country_flag = pfp_is_western_usa
				}

				clear_global_event_target = pfp_western_usa_winner
				set_global_flag = pfp_western_usa_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Northeastern (P20) + Midwestern (P18) + Southern (P26) -> USA
	# ----------------------------------------------------------------
	pfp_merge_usa = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_usa_merged }
			OR = {
				P20 = { is_subject_of = ROOT }
				P18 = { is_subject_of = ROOT }
				P26 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_not_yet_formed_tt
				NOT = { has_global_flag = pfp_usa_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_puppets_ready_tt
				P20 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P18 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P26 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_usa_effect_form
			custom_effect_tooltip = pfp_merge_usa_effect_annex
			custom_effect_tooltip = pfp_merge_usa_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories
				P20 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
				}
				P18 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
				}
				P26 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
				}

				# P20 Northeastern wins by default.
				# P18 wins if it strictly has a higher score than P20.
				# P26 wins if it strictly has a higher score than the current leader.
				P20 = { set_country_flag = pfp_usa_winner }
				P18 = {
					if = {
						limit = { check_variable = { pfp_fac > P20.pfp_fac } }
						set_country_flag = pfp_usa_winner
						P20 = { clr_country_flag = pfp_usa_winner }
					}
				}
				P26 = {
					if = {
						limit = {
							OR = {
								AND = {
									P20 = { has_country_flag = pfp_usa_winner }
									check_variable = { pfp_fac > P20.pfp_fac }
								}
								AND = {
									P18 = { has_country_flag = pfp_usa_winner }
									check_variable = { pfp_fac > P18.pfp_fac }
								}
							}
						}
						set_country_flag = pfp_usa_winner
						P20 = { clr_country_flag = pfp_usa_winner }
						P18 = { clr_country_flag = pfp_usa_winner }
					}
				}

				# P20 Northeastern U.S. wins
				if = {
					limit = { P20 = { has_country_flag = pfp_usa_winner } }

					P20 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P18.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P20.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P26.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P20.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P18  has_contested_owner = P18 } }
						add_core_of = P20
						add_contested_owner = P20
						remove_core_of = P18
						remove_claim_by = P18
						remove_contested_owner = P18
					}
					every_state = {
						limit = { OR = { is_owned_by = P26  has_contested_owner = P26 } }
						add_core_of = P20
						add_contested_owner = P20
						remove_core_of = P26
						remove_claim_by = P26
						remove_contested_owner = P26
					}

					P20 = {
						annex_country = { target = P18 transfer_troops = yes }
						annex_country = { target = P26 transfer_troops = yes }
						set_cosmetic_tag = PFP_EASTERN_USA
						set_country_flag = pfp_is_united_states
					}
				}

				# P18 Midwestern U.S. wins
				else_if = {
					limit = { P18 = { has_country_flag = pfp_usa_winner } }

					P18 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P20.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P18.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P26.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P18.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P20  has_contested_owner = P20 } }
						add_core_of = P18
						add_contested_owner = P18
						remove_core_of = P20
						remove_claim_by = P20
						remove_contested_owner = P20
					}
					every_state = {
						limit = { OR = { is_owned_by = P26  has_contested_owner = P26 } }
						add_core_of = P18
						add_contested_owner = P18
						remove_core_of = P26
						remove_claim_by = P26
						remove_contested_owner = P26
					}

					P18 = {
						annex_country = { target = P20 transfer_troops = yes }
						annex_country = { target = P26 transfer_troops = yes }
						set_cosmetic_tag = PFP_EASTERN_USA
						set_country_flag = pfp_is_united_states
					}
				}

				# P26 Southern U.S. wins
				else = {

					P26 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P20.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P26.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P18.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P26.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P20  has_contested_owner = P20 } }
						add_core_of = P26
						add_contested_owner = P26
						remove_core_of = P20
						remove_claim_by = P20
						remove_contested_owner = P20
					}
					every_state = {
						limit = { OR = { is_owned_by = P18  has_contested_owner = P18 } }
						add_core_of = P26
						add_contested_owner = P26
						remove_core_of = P18
						remove_claim_by = P18
						remove_contested_owner = P18
					}

					P26 = {
						annex_country = { target = P20 transfer_troops = yes }
						annex_country = { target = P18 transfer_troops = yes }
						set_cosmetic_tag = PFP_EASTERN_USA
						set_country_flag = pfp_is_united_states
					}
				}

				set_global_flag = pfp_usa_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Western U.S.A. + Eastern U.S.A. -> United States (PFP_USA)
	# ----------------------------------------------------------------
	pfp_merge_united_states = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_western_usa_merged
			has_global_flag = pfp_usa_merged
			NOT = { has_global_flag = pfp_united_states_merged }
			OR = {
				any_subject_country = { has_country_flag = pfp_is_western_usa }
				any_subject_country = { has_country_flag = pfp_is_united_states }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_united_states_not_yet_formed_tt
				NOT = { has_global_flag = pfp_united_states_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_western_usa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_western_usa }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_eastern_usa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_united_states }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_subject_is_military_occupation_tt
				any_subject_country = {
					has_country_flag = pfp_is_western_usa
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_subject_is_military_occupation_tt
				any_subject_country = {
					has_country_flag = pfp_is_united_states
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_puppet_must_core_owned_territories_tt
				NOT = {
					any_subject_country = {
						has_country_flag = pfp_is_western_usa
						any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } }
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_puppet_must_core_owned_territories_tt
				NOT = {
					any_subject_country = {
						has_country_flag = pfp_is_united_states
						any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } }
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_united_states_effect_form
			custom_effect_tooltip = pfp_merge_united_states_effect_annex
			custom_effect_tooltip = pfp_merge_united_states_effect_cores
			hidden_effect = {
				# Score both and initialize winner as western usa
				every_subject_country = {
					limit = { has_country_flag = pfp_is_western_usa }
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_usa_west
					save_global_event_target_as = pfp_united_states_winner
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_united_states }
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_usa_east
				}

				# Eastern USA challenges western for winner
				if = {
					limit = { check_variable = { event_target:pfp_usa_east.pfp_fac > event_target:pfp_united_states_winner.pfp_fac } }
					event_target:pfp_usa_east = { save_global_event_target_as = pfp_united_states_winner }
				}

				# Transfer western usa states if it is not the winner
				if = {
					limit = { NOT = { event_target:pfp_united_states_winner = { has_country_flag = pfp_is_western_usa } } }
					every_state = {
						limit = { OR = { is_owned_by = event_target:pfp_usa_west has_contested_owner = event_target:pfp_usa_west } }
						add_core_of = event_target:pfp_united_states_winner
						add_contested_owner = event_target:pfp_united_states_winner
						remove_core_of = event_target:pfp_usa_west
						remove_claim_by = event_target:pfp_usa_west
						remove_contested_owner = event_target:pfp_usa_west
					}
					event_target:pfp_united_states_winner = {
						for_each_loop = { array = event_target:pfp_usa_west.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
						annex_country = { target = event_target:pfp_usa_west transfer_troops = yes }
					}
				}

				# Transfer eastern usa states if it is not the winner
				if = {
					limit = { NOT = { event_target:pfp_united_states_winner = { has_country_flag = pfp_is_united_states } } }
					every_state = {
						limit = { OR = { is_owned_by = event_target:pfp_usa_east has_contested_owner = event_target:pfp_usa_east } }
						add_core_of = event_target:pfp_united_states_winner
						add_contested_owner = event_target:pfp_united_states_winner
						remove_core_of = event_target:pfp_usa_east
						remove_claim_by = event_target:pfp_usa_east
						remove_contested_owner = event_target:pfp_usa_east
					}
					event_target:pfp_united_states_winner = {
						for_each_loop = { array = event_target:pfp_usa_east.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
						annex_country = { target = event_target:pfp_usa_east transfer_troops = yes }
					}
				}

				event_target:pfp_united_states_winner = {
					set_cosmetic_tag = PFP_USA
					set_country_flag = pfp_is_usa
					set_country_flag = pfp_is_united_states
				}

				clear_global_event_target = pfp_usa_west
				clear_global_event_target = pfp_usa_east
				clear_global_event_target = pfp_united_states_winner
				set_global_flag = pfp_united_states_merged
			}
		}

		ai_will_do = { factor = 0 }
	}


}

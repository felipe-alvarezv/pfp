###########################################################################
# MERGE DECISIONS - UNITED STATES
# Allows the overlord to merge American puppet states into a unified USA.
#
# Decision 1: pfp_merge_usa
#   Requires Northeastern U.S. (P20), Midwestern U.S. (P18), and
#   Southern U.S. (P26) as PFP subjects.
#   The puppet with the most total factories + scaled manpower annexes
#   the other two and becomes the United States.
#
# Decision 2: pfp_merge_usa_rockies
#   Follow-up available after the USA is formed.
#   Requires Rockies U.S. (P23) as a PFP subject.
#
# Decision 3: pfp_merge_usa_pacific
#   Follow-up available after the USA is formed.
#   Requires Pacific U.S. (P21) as a PFP subject.
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Northeastern (P20) + Midwestern (P18) + Southern (P26) -> USA
	# ----------------------------------------------------------------
	pfp_merge_usa = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_usa_merged }
			OR = {
				P20 = { is_subject_of = ROOT }
				P18 = { is_subject_of = ROOT }
				P26 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_not_yet_formed_tt
				NOT = { has_global_flag = pfp_usa_merged }
			}
			P20 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P18 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P26 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_usa_effect_form
			custom_effect_tooltip = pfp_merge_usa_effect_annex
			custom_effect_tooltip = pfp_merge_usa_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories + (manpower / 100000)
				P20 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					set_variable = { pfp_mp_temp = manpower }
					divide_variable = { pfp_mp_temp = 100000 }
					add_to_variable = { pfp_fac = pfp_mp_temp }
				}
				P18 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					set_variable = { pfp_mp_temp = manpower }
					divide_variable = { pfp_mp_temp = 100000 }
					add_to_variable = { pfp_fac = pfp_mp_temp }
				}
				P26 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					set_variable = { pfp_mp_temp = manpower }
					divide_variable = { pfp_mp_temp = 100000 }
					add_to_variable = { pfp_fac = pfp_mp_temp }
				}

				# P20 Northeastern wins by default.
				# P18 wins if it strictly has a higher score than P20.
				# P26 wins if it strictly has a higher score than the current leader.
				P20 = { set_country_flag = pfp_usa_winner }
				P18 = {
					if = {
						limit = { check_variable = { pfp_fac > P20.pfp_fac } }
						set_country_flag = pfp_usa_winner
						P20 = { clr_country_flag = pfp_usa_winner }
					}
				}
				P26 = {
					if = {
						limit = {
							OR = {
								AND = {
									P20 = { has_country_flag = pfp_usa_winner }
									check_variable = { pfp_fac > P20.pfp_fac }
								}
								AND = {
									P18 = { has_country_flag = pfp_usa_winner }
									check_variable = { pfp_fac > P18.pfp_fac }
								}
							}
						}
						set_country_flag = pfp_usa_winner
						P20 = { clr_country_flag = pfp_usa_winner }
						P18 = { clr_country_flag = pfp_usa_winner }
					}
				}

				# P20 Northeastern U.S. wins
				if = {
					limit = { P20 = { has_country_flag = pfp_usa_winner } }

					P20 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P18.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P20.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P26.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P20.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P18  has_contested_owner = P18 } }
						add_core_of = P20
						add_contested_owner = P20
						remove_core_of = P18
						remove_claim_by = P18
						remove_contested_owner = P18
					}
					every_state = {
						limit = { OR = { is_owned_by = P26  has_contested_owner = P26 } }
						add_core_of = P20
						add_contested_owner = P20
						remove_core_of = P26
						remove_claim_by = P26
						remove_contested_owner = P26
					}

					P20 = {
						annex_country = { target = P18 transfer_troops = yes }
						annex_country = { target = P26 transfer_troops = yes }
						set_cosmetic_tag = PFP_USA
						set_country_flag = pfp_is_united_states
					}
				}

				# P18 Midwestern U.S. wins
				else_if = {
					limit = { P18 = { has_country_flag = pfp_usa_winner } }

					P18 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P20.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P18.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P26.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P18.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P20  has_contested_owner = P20 } }
						add_core_of = P18
						add_contested_owner = P18
						remove_core_of = P20
						remove_claim_by = P20
						remove_contested_owner = P20
					}
					every_state = {
						limit = { OR = { is_owned_by = P26  has_contested_owner = P26 } }
						add_core_of = P18
						add_contested_owner = P18
						remove_core_of = P26
						remove_claim_by = P26
						remove_contested_owner = P26
					}

					P18 = {
						annex_country = { target = P20 transfer_troops = yes }
						annex_country = { target = P26 transfer_troops = yes }
						set_cosmetic_tag = PFP_USA
						set_country_flag = pfp_is_united_states
					}
				}

				# P26 Southern U.S. wins
				else = {

					P26 = { clr_country_flag = pfp_usa_winner }

					for_each_loop = {
						array = P20.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P26.pfp_possible_cores = pfp_temp_core_id }
					}
					for_each_loop = {
						array = P18.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { P26.pfp_possible_cores = pfp_temp_core_id }
					}

					every_state = {
						limit = { OR = { is_owned_by = P20  has_contested_owner = P20 } }
						add_core_of = P26
						add_contested_owner = P26
						remove_core_of = P20
						remove_claim_by = P20
						remove_contested_owner = P20
					}
					every_state = {
						limit = { OR = { is_owned_by = P18  has_contested_owner = P18 } }
						add_core_of = P26
						add_contested_owner = P26
						remove_core_of = P18
						remove_claim_by = P18
						remove_contested_owner = P18
					}

					P26 = {
						annex_country = { target = P20 transfer_troops = yes }
						annex_country = { target = P18 transfer_troops = yes }
						set_cosmetic_tag = PFP_USA
						set_country_flag = pfp_is_united_states
					}
				}

				set_global_flag = pfp_usa_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed USA + Rockies U.S. (P23) -> absorb Rockies into USA
	# Follow-up: only available after pfp_merge_usa has fired
	# ----------------------------------------------------------------
	pfp_merge_usa_rockies = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_usa_merged
			NOT = { has_global_flag = pfp_usa_absorbed_rockies }
			P23 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_rockies_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_usa_absorbed_rockies }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_united_states }
			}
			P23 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_usa_rockies_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_united_states }
					save_global_event_target_as = pfp_usa_country
					for_each_loop = {
						array = P23.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = P23  has_contested_owner = P23 } }
					add_core_of = event_target:pfp_usa_country
					add_contested_owner = event_target:pfp_usa_country
					remove_core_of = P23
					remove_claim_by = P23
					remove_contested_owner = P23
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_united_states }
					annex_country = { target = P23 transfer_troops = yes }
				}

				clear_global_event_target = pfp_usa_country

				set_global_flag = pfp_usa_absorbed_rockies
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed USA + Pacific U.S. (P21) -> absorb Pacific into USA
	# Follow-up: only available after pfp_merge_usa has fired
	# ----------------------------------------------------------------
	pfp_merge_usa_pacific = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_usa_merged
			has_global_flag = pfp_usa_absorbed_rockies
			NOT = { has_global_flag = pfp_usa_absorbed_pacific }
			P21 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_pacific_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_usa_absorbed_pacific }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_united_states }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_usa_pacific_requires_rockies_tt
				has_global_flag = pfp_usa_absorbed_rockies
			}
			P21 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_usa_pacific_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_united_states }
					save_global_event_target_as = pfp_usa_country
					for_each_loop = {
						array = P21.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = P21  has_contested_owner = P21 } }
					add_core_of = event_target:pfp_usa_country
					add_contested_owner = event_target:pfp_usa_country
					remove_core_of = P21
					remove_claim_by = P21
					remove_contested_owner = P21
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_united_states }
					annex_country = { target = P21 transfer_troops = yes }
				}

				clear_global_event_target = pfp_usa_country

				set_global_flag = pfp_usa_absorbed_pacific
			}
		}

		ai_will_do = { factor = 0 }
	}

}

pfp_category = {
	pfp_northern_rhodesia = {
		allowed = {
			NOT = {
				original_tag = ZAM #Zambia
			}
		}
		available = {
			is_subject = no
			#Northern Rhodesia
			custom_trigger_tooltip = {
				tooltip = pfp_northern_rhodesia_required_states_northeast_rhodesia
				has_full_control_of_state = 771 #Northeast Rhodesia
			}
			custom_trigger_tooltip = {
				tooltip = pfp_northern_rhodesia_required_states_barotziland
				has_full_control_of_state = 981 #Barotziland
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 771 #Northeast Rhodesia
				has_full_control_of_state = 981 #Barotziland
			}
			NOT = { has_global_flag = pfp_northern_rhodesia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 771 #Northeast Rhodesia
				state = 981 #Barotziland
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_northern_rhodesia_transfer_capital
			custom_effect_tooltip = pfp_northern_rhodesia_puppet
			custom_effect_tooltip = pfp_northern_rhodesia_occupation_type
			custom_effect_tooltip = pfp_northern_rhodesia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				771 = { transfer_state_to = Q63 } #Northeast Rhodesia

				#Formable puppet becomes a puppet of ROOT
				puppet = Q63

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q63 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_northern_rhodesia = yes

				#Transfer all technology from ROOT to formable puppet
				Q63 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_northern_rhodesia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

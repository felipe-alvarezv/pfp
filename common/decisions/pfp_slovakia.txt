pfp_category = {
	pfp_slovakia = {
		allowed = {
			NOT = {
				original_tag = SLO #Slovakia
			}
		}
		available = {
			is_subject = no
			#Slovakia
			custom_trigger_tooltip = {
				tooltip = pfp_slovakia_required_states_western_slovakia
				has_full_control_of_state = 70 #Western Slovakia
			}
			custom_trigger_tooltip = {
				tooltip = pfp_slovakia_required_states_eastern_slovakia
				has_full_control_of_state = 71 #Eastern Slovakia
			}
			custom_trigger_tooltip = {
				tooltip = pfp_slovakia_required_states_southern_slovakia
				has_full_control_of_state = 664 #Southern Slovakia
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 70 #Western Slovakia
				has_full_control_of_state = 71 #Eastern Slovakia
				has_full_control_of_state = 664 #Southern Slovakia
			}
			NOT = { has_global_flag = pfp_slovakia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 70 #Western Slovakia
				state = 71 #Eastern Slovakia
				state = 664 #Southern Slovakia
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_slovakia_transfer_capital
			custom_effect_tooltip = pfp_slovakia_puppet
			custom_effect_tooltip = pfp_slovakia_occupation_type
			custom_effect_tooltip = pfp_slovakia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				70 = { transfer_state_to = Q78 } #Western Slovakia

				#Formable puppet becomes a puppet of ROOT
				puppet = Q78

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q78 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_slovakia = yes

				#Transfer all technology from ROOT to formable puppet
				Q78 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_slovakia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

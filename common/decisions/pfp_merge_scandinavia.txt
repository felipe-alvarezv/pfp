###########################################################################
# MERGE DECISIONS - SCANDINAVIA
# Allows the overlord to merge Scandinavian puppet states into a unified
# Scandinavia.
#
# Decision 1: pfp_merge_scandinavia
#   Requires the 3 core Scandinavian puppets:
#   Norway (P65), Sweden (P74), Denmark (P61).
#   The puppet with the most total factories becomes Scandinavia.
#
# Follow-up decisions (available after Scandinavia is formed):
#   pfp_merge_scandinavia_iceland   - Absorbs Iceland (P83)
#   pfp_merge_scandinavia_greenland - Absorbs Greenland (P85)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Norway (P65) + Sweden (P74) + Denmark (P61) -> Scandinavia
	# ----------------------------------------------------------------
	pfp_merge_scandinavia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_scandinavia_merged }
			OR = {
				P65 = { is_subject_of = ROOT }
				P74 = { is_subject_of = ROOT }
				P61 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_scandinavia_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_puppets_ready_tt
				P65 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P74 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P61 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_scandinavia_effect_form
			custom_effect_tooltip = pfp_merge_scandinavia_effect_annex
			custom_effect_tooltip = pfp_merge_scandinavia_effect_cores
			hidden_effect = {
				# P65 Norway initializes as default winner.
				# Each subsequent tag wins if it strictly has more factories.
				P65 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_scandinavia_winner
				}
				P74 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_scandinavia_winner.pfp_fac } }
						save_global_event_target_as = pfp_scandinavia_winner
					}
				}
				P61 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_scandinavia_winner.pfp_fac } }
						save_global_event_target_as = pfp_scandinavia_winner
					}
				}

				# Transfer state cores/ownership from losers to winner
				if = {
					limit = { NOT = { event_target:pfp_scandinavia_winner = { tag = P65 } } }
					every_state = {
						limit = { OR = { is_owned_by = P65  has_contested_owner = P65 } }
						add_core_of = event_target:pfp_scandinavia_winner
						add_contested_owner = event_target:pfp_scandinavia_winner
						remove_core_of = P65
						remove_claim_by = P65
						remove_contested_owner = P65
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_scandinavia_winner = { tag = P74 } } }
					every_state = {
						limit = { OR = { is_owned_by = P74  has_contested_owner = P74 } }
						add_core_of = event_target:pfp_scandinavia_winner
						add_contested_owner = event_target:pfp_scandinavia_winner
						remove_core_of = P74
						remove_claim_by = P74
						remove_contested_owner = P74
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_scandinavia_winner = { tag = P61 } } }
					every_state = {
						limit = { OR = { is_owned_by = P61  has_contested_owner = P61 } }
						add_core_of = event_target:pfp_scandinavia_winner
						add_contested_owner = event_target:pfp_scandinavia_winner
						remove_core_of = P61
						remove_claim_by = P61
						remove_contested_owner = P61
					}
				}

				# Transfer pfp_possible_cores arrays, annex losers, set tag and flag
				event_target:pfp_scandinavia_winner = {
					if = { limit = { NOT = { tag = P65 } } for_each_loop = { array = P65.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P74 } } for_each_loop = { array = P74.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P61 } } for_each_loop = { array = P61.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P65 } } annex_country = { target = P65 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P74 } } annex_country = { target = P74 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P61 } } annex_country = { target = P61 transfer_troops = yes } }
					set_cosmetic_tag = PFP_SCANDINAVIA
					set_country_flag = pfp_is_scandinavia
				}

				clear_global_event_target = pfp_scandinavia_winner
				set_global_flag = pfp_scandinavia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Scandinavia + Iceland (P83) -> absorb Iceland
	# ----------------------------------------------------------------
	pfp_merge_scandinavia_iceland = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_scandinavia_merged
			NOT = { has_global_flag = pfp_scandinavia_absorbed_iceland }
			P83 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_iceland_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_scandinavia_absorbed_iceland }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_scandinavia }
			}
			P83 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_scandinavia_iceland_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_scandinavia }
					save_global_event_target_as = pfp_scandinavia_country
					for_each_loop = {
						array = P83.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = P83  has_contested_owner = P83 } }
					add_core_of = event_target:pfp_scandinavia_country
					add_contested_owner = event_target:pfp_scandinavia_country
					remove_core_of = P83
					remove_claim_by = P83
					remove_contested_owner = P83
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_scandinavia }
					annex_country = { target = P83 transfer_troops = yes }
				}
				clear_global_event_target = pfp_scandinavia_country
				set_global_flag = pfp_scandinavia_absorbed_iceland
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Scandinavia + Greenland (P85) -> absorb Greenland
	# ----------------------------------------------------------------
	pfp_merge_scandinavia_greenland = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_scandinavia_merged
			NOT = { has_global_flag = pfp_scandinavia_absorbed_greenland }
			P85 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_greenland_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_scandinavia_absorbed_greenland }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_scandinavia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_scandinavia }
			}
			P85 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_scandinavia_greenland_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_scandinavia }
					save_global_event_target_as = pfp_scandinavia_country
					for_each_loop = {
						array = P85.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = P85  has_contested_owner = P85 } }
					add_core_of = event_target:pfp_scandinavia_country
					add_contested_owner = event_target:pfp_scandinavia_country
					remove_core_of = P85
					remove_claim_by = P85
					remove_contested_owner = P85
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_scandinavia }
					annex_country = { target = P85 transfer_troops = yes }
				}
				clear_global_event_target = pfp_scandinavia_country
				set_global_flag = pfp_scandinavia_absorbed_greenland
			}
		}

		ai_will_do = { factor = 0 }
	}

}

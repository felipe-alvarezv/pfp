# MERGE DECISIONS - SOUTH AFRICA

# unified South Africa.
#
# Decision 1: pfp_merge_south_africa
#   Requires all 8 Southern African puppets:
#   South Africa (P36), Angola (Q67), Mozambique (Q68),
#   Southern Rhodesia (Q69), Bechuanaland (Q70),
#   South West Africa (Q71), Northern Rhodesia (Q63), Nyasaland (Q64).
#   The puppet with the most total factories becomes South Africa.
#
# Follow-up decisions (available after South Africa is formed):
#   pfp_merge_south_africa_madagascar - Absorbs Madagascar (Q72)
###########################################################################

						add_contested_owner = event_target:pfp_south_africa_winner
						remove_core_of = Q63
						remove_claim_by = Q63
						remove_contested_owner = Q63

					every_state = {

						add_contested_owner = event_target:pfp_south_africa_winner

	pfp_merge_south_africa_madagascar = {

			is_ai = no

				tooltip = pfp_merge_south_africa_is_formed_tt

			}
			Q72 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_south_africa_madagascar_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_south_africa }
					save_global_event_target_as = pfp_south_africa_country
					for_each_loop = {
						array = Q72.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = Q72  has_contested_owner = Q72 } }
					add_core_of = event_target:pfp_south_africa_country
					add_contested_owner = event_target:pfp_south_africa_country
					remove_core_of = Q72
					remove_claim_by = Q72
					remove_contested_owner = Q72
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_south_africa }
					annex_country = { target = Q72 transfer_troops = yes }
				}
				clear_global_event_target = pfp_south_africa_country
				set_global_flag = pfp_south_africa_absorbed_madagascar
			}
		}

		ai_will_do = { factor = 0 }
	}

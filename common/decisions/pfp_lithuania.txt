pfp_category = {
	pfp_lithuania = {
		allowed = {
			NOT = {
				original_tag = LIT #Lithuania
			}
		}
		available = {
			is_subject = no
			#Kaunas
			custom_trigger_tooltip = {
				tooltip = pfp_lithuania_required_states_kaunas
				has_full_control_of_state = 11 #Kaunas
			}
			#Suduva
			custom_trigger_tooltip = {
				tooltip = pfp_lithuania_required_states_suduva
				has_full_control_of_state = 814 #Suduva
			}
			#Aukstaitija
			custom_trigger_tooltip = {
				tooltip = pfp_lithuania_required_states_aukstaitija
				has_full_control_of_state = 815 #Aukstaitija
			}
			#Zemaitija
			custom_trigger_tooltip = {
				tooltip = pfp_lithuania_required_states_zemaitija
				has_full_control_of_state = 189 #Zemaitija
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 11 #Kaunas
				has_full_control_of_state = 814 #Suduva
				has_full_control_of_state = 815 #Aukstaitija
				has_full_control_of_state = 189 #Zemaitija
			}
			NOT = { has_global_flag = pfp_lithuania_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 11 #Kaunas
				state = 814 #Suduva
				state = 815 #Aukstaitija
				state = 189 #Zemaitija
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_lithuania_transfer_capital
			custom_effect_tooltip = pfp_lithuania_puppet
			custom_effect_tooltip = pfp_lithuania_occupation_type
			custom_effect_tooltip = pfp_lithuania_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				11 = { transfer_state_to = R11 } #Kaunas

				#Formable puppet becomes a puppet of ROOT
				puppet = R11

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R11 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_lithuania = yes

				#Transfer all technology from ROOT to formable puppet
				R11 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_lithuania_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

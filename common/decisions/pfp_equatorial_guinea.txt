pfp_category = {
	pfp_equatorial_guinea = {
		allowed = {
			NOT = {
				original_tag = EQG #Equatorial Guinea
			}
		}
		available = {
			is_subject = no
			#Equatorial Guinea
			custom_trigger_tooltip = {
				tooltip = pfp_equatorial_guinea_required_states_equatorial_guinea
				has_full_control_of_state = 297 #Equatorial Guinea
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 297 #Equatorial Guinea
			}
			NOT = { has_global_flag = pfp_equatorial_guinea_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 297 #Equatorial Guinea
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_equatorial_guinea_transfer_capital
			custom_effect_tooltip = pfp_equatorial_guinea_puppet
			custom_effect_tooltip = pfp_equatorial_guinea_occupation_type
			custom_effect_tooltip = pfp_equatorial_guinea_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				297 = { transfer_state_to = Q54 } #Equatorial Guinea

				#Formable puppet becomes a puppet of ROOT
				puppet = Q54

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q54 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_equatorial_guinea = yes

				#Transfer all technology from ROOT to formable puppet
				Q54 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_equatorial_guinea_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

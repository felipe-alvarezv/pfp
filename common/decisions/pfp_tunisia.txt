pfp_category = {
	pfp_tunisia = {
		allowed = {
			NOT = {
				original_tag = TUN #Tunisia
			}
		}
		available = {
			is_subject = no
			#Tunisia
			custom_trigger_tooltip = {
				tooltip = pfp_tunisia_required_states_tunisia
				has_full_control_of_state = 458 #Tunisia
			}
			#Gabès
			custom_trigger_tooltip = {
				tooltip = pfp_tunisia_required_states_gabes
				has_full_control_of_state = 665 #Gabès
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 458 #Tunisia
				has_full_control_of_state = 665 #Gabès
			}
			NOT = { has_global_flag = pfp_tunisia_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 458 #Tunisia
				state = 665 #Gabès
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_tunisia_transfer_capital
			custom_effect_tooltip = pfp_tunisia_puppet
			custom_effect_tooltip = pfp_tunisia_occupation_type
			custom_effect_tooltip = pfp_tunisia_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				458 = { transfer_state_to = Q28 } #Tunisia

				#Formable puppet becomes a puppet of ROOT
				puppet = Q28

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q28 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_tunisia = yes

				#Transfer all technology from ROOT to formable puppet
				Q28 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_tunisia_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

###########################################################################
# MERGE DECISIONS - WEST INDIES
# Allows the overlord to merge West Indian puppet states into a unified
# West Indies.
#
# Decision 1: pfp_merge_west_indies
#   Requires Cuba (Q99), Dominican Republic (Q98), and Haiti (Q97).
#   The puppet with the most total factories becomes the West Indies.
#
# Follow-up decisions (available after West Indies is formed):
#   pfp_merge_west_indies_jamaica        - Absorbs Jamaica (Q96)
#   pfp_merge_west_indies_bahamas        - Absorbs Bahamas (R01)
#   pfp_merge_west_indies_puerto_rico    - Absorbs Puerto Rico (R02)
#   pfp_merge_west_indies_french_antilles - Absorbs French Antilles (R03)
#   pfp_merge_west_indies_british_antilles - Absorbs British Antilles (R04)
#   pfp_merge_west_indies_dutch_antilles - Absorbs Dutch Antilles (R05)
#   pfp_merge_west_indies_trinidad       - Absorbs Trinidad & Tobago (R06)
#   pfp_merge_west_indies_bermuda        - Absorbs Bermuda (R07)
#   pfp_merge_west_indies_belize         - Absorbs Belize (R08)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Cuba (Q99) + Dominican Republic (Q98) + Haiti (Q97) -> West Indies
	# ----------------------------------------------------------------
	pfp_merge_west_indies = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_west_indies_merged }
			OR = {
				Q97 = { is_subject_of = ROOT }
				Q98 = { is_subject_of = ROOT }
				Q99 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_west_indies_not_yet_formed_tt
				NOT = { has_global_flag = pfp_west_indies_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_west_indies_puppets_ready_tt
				Q97 = { is_subject_of = ROOT }
				Q98 = { is_subject_of = ROOT }
				Q99 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				Q97 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q98 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				Q99 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				Q97 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q98 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				Q99 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_west_indies_effect_form
			custom_effect_tooltip = pfp_merge_west_indies_effect_annex
			custom_effect_tooltip = pfp_merge_west_indies_effect_cores
			hidden_effect = {
				# Each subsequent puppet wins if it strictly has more factories.
				Q97 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_west_indies_winner
				}
				Q98 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_west_indies_winner.pfp_fac } }
						save_global_event_target_as = pfp_west_indies_winner
					}
				}
				Q99 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_west_indies_winner.pfp_fac } }
						save_global_event_target_as = pfp_west_indies_winner
					}
				}

				# Transfer state cores/ownership from losers to winner
				if = {
					limit = { NOT = { event_target:pfp_west_indies_winner = { tag = Q97 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q97  has_contested_owner = Q97 } }
						add_core_of = event_target:pfp_west_indies_winner
						add_contested_owner = event_target:pfp_west_indies_winner
						remove_core_of = Q97
						remove_claim_by = Q97
						remove_contested_owner = Q97
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_west_indies_winner = { tag = Q98 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q98  has_contested_owner = Q98 } }
						add_core_of = event_target:pfp_west_indies_winner
						add_contested_owner = event_target:pfp_west_indies_winner
						remove_core_of = Q98
						remove_claim_by = Q98
						remove_contested_owner = Q98
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_west_indies_winner = { tag = Q99 } } }
					every_state = {
						limit = { OR = { is_owned_by = Q99  has_contested_owner = Q99 } }
						add_core_of = event_target:pfp_west_indies_winner
						add_contested_owner = event_target:pfp_west_indies_winner
						remove_core_of = Q99
						remove_claim_by = Q99
						remove_contested_owner = Q99
					}
				}

				# Transfer pfp_possible_cores arrays, annex losers, set tag and flag
				event_target:pfp_west_indies_winner = {
					if = { limit = { NOT = { tag = Q97 } } for_each_loop = { array = Q97.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q98 } } for_each_loop = { array = Q98.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q99 } } for_each_loop = { array = Q99.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = Q97 } } annex_country = { target = Q97 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q98 } } annex_country = { target = Q98 transfer_troops = yes } }
					if = { limit = { NOT = { tag = Q99 } } annex_country = { target = Q99 transfer_troops = yes } }
					set_cosmetic_tag = PFP_WEST_INDIES
					set_country_flag = pfp_is_west_indies
				}

				clear_global_event_target = pfp_west_indies_winner
				set_global_flag = pfp_west_indies_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

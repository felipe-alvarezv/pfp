pfp_category = {
	pfp_morocco = {
		allowed = {
			NOT = {
				original_tag = MOR #Morocco
				original_tag = WES #Sahrawi Arab Republic
			}
		}
		available = {
			is_subject = no
			#Casablanca
			custom_trigger_tooltip = {
				tooltip = pfp_morocco_required_states_casablanca
				has_full_control_of_state = 461 #Casablanca
			}
			#Marrakech
			custom_trigger_tooltip = {
				tooltip = pfp_morocco_required_states_marrakech
				has_full_control_of_state = 462 #Marrakech
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 461 #Casablanca
				has_full_control_of_state = 462 #Marrakech
			}
			NOT = { has_global_flag = pfp_morocco_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 461 #Casablanca
				state = 462 #Marrakech
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_morocco_transfer_capital
			custom_effect_tooltip = pfp_morocco_puppet
			custom_effect_tooltip = pfp_morocco_occupation_type
			custom_effect_tooltip = pfp_morocco_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				461 = { transfer_state_to = Q26 } #Casablanca

				#Formable puppet becomes a puppet of ROOT
				puppet = Q26

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q26 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_morocco = yes

				#Transfer all technology from ROOT to formable puppet
				Q26 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_morocco_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

###########################################################################
# MERGE DECISIONS - BALTIC STATES
# Allows the overlord to merge Baltic puppet states into a unified
# Baltic States.
#
# Decision 1: pfp_merge_baltic
#   Requires Estonia (R09) and Latvia (R10) as PFP subjects.
#   The puppet with the most total factories becomes the Baltic States.
#
# Follow-up decisions (available after Baltic States is formed):
#   pfp_merge_baltic_lithuania - Absorbs Lithuania (R11)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Estonia (R09) + Latvia (R10) -> Baltic States
	# ----------------------------------------------------------------
	pfp_merge_baltic = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_baltic_merged }
			OR = {
				R09 = { is_subject_of = ROOT }
				R10 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_baltic_not_yet_formed_tt
				NOT = { has_global_flag = pfp_baltic_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_baltic_puppets_ready_tt
				R09 = { is_subject_of = ROOT }
				R10 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				R09 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				R10 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				R09 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				R10 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_baltic_effect_form
			custom_effect_tooltip = pfp_merge_baltic_effect_annex
			custom_effect_tooltip = pfp_merge_baltic_effect_cores
			hidden_effect = {
				# R10 Latvia wins if it strictly has more factories.
				R09 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_baltic_winner
				}
				R10 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_baltic_winner.pfp_fac } }
						save_global_event_target_as = pfp_baltic_winner
					}
				}

				# Transfer state cores/ownership from loser to winner
				if = {
					limit = { NOT = { event_target:pfp_baltic_winner = { tag = R09 } } }
					every_state = {
						limit = { OR = { is_owned_by = R09  has_contested_owner = R09 } }
						add_core_of = event_target:pfp_baltic_winner
						add_contested_owner = event_target:pfp_baltic_winner
						remove_core_of = R09
						remove_claim_by = R09
						remove_contested_owner = R09
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_baltic_winner = { tag = R10 } } }
					every_state = {
						limit = { OR = { is_owned_by = R10  has_contested_owner = R10 } }
						add_core_of = event_target:pfp_baltic_winner
						add_contested_owner = event_target:pfp_baltic_winner
						remove_core_of = R10
						remove_claim_by = R10
						remove_contested_owner = R10
					}
				}

				# Transfer pfp_possible_cores arrays, annex loser, set tag and flag
				event_target:pfp_baltic_winner = {
					if = { limit = { NOT = { tag = R09 } } for_each_loop = { array = R09.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R10 } } for_each_loop = { array = R10.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R09 } } annex_country = { target = R09 transfer_troops = yes } }
					if = { limit = { NOT = { tag = R10 } } annex_country = { target = R10 transfer_troops = yes } }
					set_cosmetic_tag = PFP_BALTIC
					set_country_flag = pfp_is_baltic
				}

				clear_global_event_target = pfp_baltic_winner
				set_global_flag = pfp_baltic_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Baltic States + Lithuania (R11) -> absorb Lithuania
	# ----------------------------------------------------------------
	pfp_merge_baltic_lithuania = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_baltic_merged
			NOT = { has_global_flag = pfp_baltic_absorbed_lithuania }
			R11 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_baltic_lithuania_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_baltic_absorbed_lithuania }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_baltic_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_baltic }
			}
			R11 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
					has_autonomy_state = pfp_autonomy_military_occupation
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_baltic_lithuania_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_baltic }
					save_global_event_target_as = pfp_baltic_country
					for_each_loop = {
						array = R11.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}
				every_state = {
					limit = { OR = { is_owned_by = R11  has_contested_owner = R11 } }
					add_core_of = event_target:pfp_baltic_country
					add_contested_owner = event_target:pfp_baltic_country
					remove_core_of = R11
					remove_claim_by = R11
					remove_contested_owner = R11
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_baltic }
					annex_country = { target = R11 transfer_troops = yes }
				}
				clear_global_event_target = pfp_baltic_country
				set_global_flag = pfp_baltic_absorbed_lithuania
			}
		}

		ai_will_do = { factor = 0 }
	}

}

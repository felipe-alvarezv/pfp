#      Follow-ups: P41 Northern Caucasus, Q76 Karelia, Q77 Murmansk
#   2. pfp_merge_eastern_russia: P44+P45+P46 -> PFP_EASTERN_RUSSIA
#   3. pfp_merge_russia: PFP_WESTERN_RUSSIA + PFP_EASTERN_RUSSIA -> PFP_RUSSIA

			}
			P39 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt

						has_autonomy_state = pfp_autonomy_military_occupation

					tooltip = pfp_merge_puppet_must_core_owned_territories_tt

						any_state_in = {

					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}

					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores

				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted

				# Initialize winner with P39, then linear challenge
				P39 = { save_global_event_target_as = pfp_western_russia_winner }
				if = { limit = { check_variable = { P40.pfp_fac > event_target:pfp_western_russia_winner.pfp_fac } } P40 = { save_global_event_target_as = pfp_western_russia_winner } }

				if = { limit = { check_variable = { P43.pfp_fac > event_target:pfp_western_russia_winner.pfp_fac } } P43 = { save_global_event_target_as = pfp_western_russia_winner } }

				# Transfer state cores/ownership from non-winners to winner
				if = {
					limit = { NOT = { event_target:pfp_western_russia_winner = { tag = P39 } } }
					every_state = { limit = { OR = { is_owned_by = P39 has_contested_owner = P39 } } add_core_of = event_target:pfp_western_russia_winner add_contested_owner = event_target:pfp_western_russia_winner remove_core_of = P39 remove_claim_by = P39 remove_contested_owner = P39 }
				}
				if = {
					limit = { NOT = { event_target:pfp_western_russia_winner = { tag = P40 } } }
					every_state = { limit = { OR = { is_owned_by = P40 has_contested_owner = P40 } } add_core_of = event_target:pfp_western_russia_winner add_contested_owner = event_target:pfp_western_russia_winner remove_core_of = P40 remove_claim_by = P40 remove_contested_owner = P40 }
				}
				if = {
					limit = { NOT = { event_target:pfp_western_russia_winner = { tag = P42 } } }
					every_state = { limit = { OR = { is_owned_by = P42 has_contested_owner = P42 } } add_core_of = event_target:pfp_western_russia_winner add_contested_owner = event_target:pfp_western_russia_winner remove_core_of = P42 remove_claim_by = P42 remove_contested_owner = P42 }
				}
				if = {
					limit = { NOT = { event_target:pfp_western_russia_winner = { tag = P43 } } }
					every_state = { limit = { OR = { is_owned_by = P43 has_contested_owner = P43 } } add_core_of = event_target:pfp_western_russia_winner add_contested_owner = event_target:pfp_western_russia_winner remove_core_of = P43 remove_claim_by = P43 remove_contested_owner = P43 }
				}

				# Transfer pfp_possible_cores arrays, annex non-winners, set tag and flag
				event_target:pfp_western_russia_winner = {
					if = { limit = { NOT = { tag = P39 } } for_each_loop = { array = P39.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P40 } } for_each_loop = { array = P40.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P42 } } for_each_loop = { array = P42.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P43 } } for_each_loop = { array = P43.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P39 } } annex_country = { target = P39 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P40 } } annex_country = { target = P40 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P42 } } annex_country = { target = P42 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P43 } } annex_country = { target = P43 transfer_troops = yes } }
					set_cosmetic_tag = PFP_WESTERN_RUSSIA
					set_country_flag = pfp_is_western_russia
				}

				clear_global_event_target = pfp_western_russia_winner
				set_global_flag = pfp_western_russia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

			}
		}

					if = { limit = { NOT = { tag = P46 } } for_each_loop = { array = P46.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P44 } } annex_country = { target = P44 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P45 } } annex_country = { target = P45 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P46 } } annex_country = { target = P46 transfer_troops = yes } }
					set_cosmetic_tag = PFP_EASTERN_RUSSIA
					set_country_flag = pfp_is_eastern_russia
				}

				clear_global_event_target = pfp_eastern_russia_winner
				set_global_flag = pfp_eastern_russia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Western Russia + Eastern Russia -> Russia (PFP_RUSSIA)
	# ----------------------------------------------------------------
	pfp_merge_russia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_western_russia_merged
			has_global_flag = pfp_eastern_russia_merged
			NOT = { has_global_flag = pfp_russia_merged }
			OR = {
				any_subject_country = { has_country_flag = pfp_is_western_russia }
				any_subject_country = { has_country_flag = pfp_is_eastern_russia }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_russia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_russia_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_western_russia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_western_russia }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_eastern_russia_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_eastern_russia }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_subject_is_military_occupation_tt
				any_subject_country = {
					has_country_flag = pfp_is_western_russia
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_subject_is_military_occupation_tt
				any_subject_country = {
					has_country_flag = pfp_is_eastern_russia
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
			NOT = {
				any_subject_country = {
					has_country_flag = pfp_is_western_russia
					any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } }
				}
			}
			NOT = {
				any_subject_country = {
					has_country_flag = pfp_is_eastern_russia
					any_state_in = { array = THIS.pfp_possible_cores is_owned_by = PREV NOT = { is_core_of = PREV } }
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_russia_effect_form
			custom_effect_tooltip = pfp_merge_russia_effect_annex
			custom_effect_tooltip = pfp_merge_russia_effect_cores
			hidden_effect = {
				# Score both formables and initialize winner as western russia
				every_subject_country = {
					limit = { has_country_flag = pfp_is_western_russia }
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					set_variable = { pfp_mp_temp = manpower }
					divide_variable = { pfp_mp_temp = 100000 }
					add_to_variable = { pfp_fac = pfp_mp_temp }
					save_global_event_target_as = pfp_russia_west
					save_global_event_target_as = pfp_russia_winner
				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_eastern_russia }
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					set_variable = { pfp_mp_temp = manpower }
					divide_variable = { pfp_mp_temp = 100000 }
					add_to_variable = { pfp_fac = pfp_mp_temp }
					save_global_event_target_as = pfp_russia_east
				}

				# Eastern Russia challenges western for winner
				if = {
					limit = { check_variable = { event_target:pfp_russia_east.pfp_fac > event_target:pfp_russia_winner.pfp_fac } }
					event_target:pfp_russia_east = { save_global_event_target_as = pfp_russia_winner }
				}

				# Transfer western russia states if it is not the winner
				if = {
					limit = { NOT = { event_target:pfp_russia_winner = { has_country_flag = pfp_is_western_russia } } }
					every_state = {
						limit = { OR = { is_owned_by = event_target:pfp_russia_west has_contested_owner = event_target:pfp_russia_west } }
						add_core_of = event_target:pfp_russia_winner
						add_contested_owner = event_target:pfp_russia_winner
						remove_core_of = event_target:pfp_russia_west
						remove_claim_by = event_target:pfp_russia_west
						remove_contested_owner = event_target:pfp_russia_west
					}
					event_target:pfp_russia_winner = {
						for_each_loop = { array = event_target:pfp_russia_west.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
						annex_country = { target = event_target:pfp_russia_west transfer_troops = yes }
					}
				}

				# Transfer eastern russia states if it is not the winner
				if = {
					limit = { NOT = { event_target:pfp_russia_winner = { has_country_flag = pfp_is_eastern_russia } } }
					every_state = {
						limit = { OR = { is_owned_by = event_target:pfp_russia_east has_contested_owner = event_target:pfp_russia_east } }
						add_core_of = event_target:pfp_russia_winner
						add_contested_owner = event_target:pfp_russia_winner
						remove_core_of = event_target:pfp_russia_east
						remove_claim_by = event_target:pfp_russia_east
						remove_contested_owner = event_target:pfp_russia_east
					}
					event_target:pfp_russia_winner = {
						for_each_loop = { array = event_target:pfp_russia_east.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
						annex_country = { target = event_target:pfp_russia_east transfer_troops = yes }
					}
				}

				event_target:pfp_russia_winner = {
					set_cosmetic_tag = PFP_RUSSIA
					set_country_flag = pfp_is_russia
				}

				clear_global_event_target = pfp_russia_west
				clear_global_event_target = pfp_russia_east
				clear_global_event_target = pfp_russia_winner
				set_global_flag = pfp_russia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

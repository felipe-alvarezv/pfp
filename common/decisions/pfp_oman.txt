	pfp_oman = {
		allowed = {
			NOT = {
				original_tag = OMA #Oman
			}
		}
		available = {
			is_subject = no
			#Muscat
			custom_trigger_tooltip = {
				tooltip = pfp_oman_required_states_muscat
				has_full_control_of_state = 294 #Muscat
			}
			#Oman
			custom_trigger_tooltip = {
				tooltip = pfp_oman_required_states_oman
				has_full_control_of_state = 1015 #Oman
			}
			#Dhofar
			custom_trigger_tooltip = {
				tooltip = pfp_oman_required_states_dhofar
				has_full_control_of_state = 1016 #Dhofar
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 294 #Muscat
				has_full_control_of_state = 1015 #Oman
				has_full_control_of_state = 1016 #Dhofar
			}
			NOT = { has_global_flag = pfp_oman_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 294 #Muscat
				state = 1015 #Oman
				state = 1016 #Dhofar
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_oman_transfer_capital
			custom_effect_tooltip = pfp_oman_puppet
			custom_effect_tooltip = pfp_oman_occupation_type

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				294 = { transfer_state_to = Q84 } #Muscat

				#Formable puppet becomes a puppet of ROOT
				puppet = Q84

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = Q84 }

				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_oman = yes

				#Transfer all technology from ROOT to formable puppet
				Q84 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_oman_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

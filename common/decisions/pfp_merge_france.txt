###########################################################################
# MERGE DECISIONS - FRANCE
# Allows the overlord to merge French puppet states into a unified France.
#
# Decision 1: pfp_merge_france
#   Requires Northern France (P62) and Southern France (P63) as PFP subjects.
#   The puppet with the most total factories + scaled manpower becomes France.
#
# Follow-up decisions (available after France is formed):
#   pfp_merge_france_alps  - Absorbs French Alps (P64)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Northern France (P62) + Southern France (P63) -> France
	# ----------------------------------------------------------------
	pfp_merge_france = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_france_merged }
			OR = {
				P62 = { is_subject_of = ROOT }
				P63 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_france_not_yet_formed_tt
				NOT = { has_global_flag = pfp_france_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_france_puppets_ready_tt
				P62 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
				P63 = {
					is_subject_of = ROOT
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_france_effect_form
			custom_effect_tooltip = pfp_merge_france_effect_annex
			custom_effect_tooltip = pfp_merge_france_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories
				# P62 Northern France initializes as default winner.
				P62 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_france_winner
				}
				P63 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_france_winner.pfp_fac } }
						save_global_event_target_as = pfp_france_winner
					}
				}

				# Transfer state cores/ownership from loser to winner
				if = {
					limit = { NOT = { event_target:pfp_france_winner = { tag = P62 } } }
					every_state = {
						limit = { OR = { is_owned_by = P62  has_contested_owner = P62 } }
						add_core_of = event_target:pfp_france_winner
						add_contested_owner = event_target:pfp_france_winner
						remove_core_of = P62
						remove_claim_by = P62
						remove_contested_owner = P62
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_france_winner = { tag = P63 } } }
					every_state = {
						limit = { OR = { is_owned_by = P63  has_contested_owner = P63 } }
						add_core_of = event_target:pfp_france_winner
						add_contested_owner = event_target:pfp_france_winner
						remove_core_of = P63
						remove_claim_by = P63
						remove_contested_owner = P63
					}
				}

				# Transfer pfp_possible_cores arrays, annex loser, set tag and flag
				event_target:pfp_france_winner = {
					if = {
						limit = { NOT = { tag = P62 } }
						for_each_loop = {
							array = P62.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = {
						limit = { NOT = { tag = P63 } }
						for_each_loop = {
							array = P63.pfp_possible_cores
							value = pfp_temp_core_id
							add_to_array = { pfp_possible_cores = pfp_temp_core_id }
						}
					}
					if = { limit = { NOT = { tag = P62 } } annex_country = { target = P62 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P63 } } annex_country = { target = P63 transfer_troops = yes } }
					set_cosmetic_tag = PFP_FRANCE
					set_country_flag = pfp_is_france
				}

				clear_global_event_target = pfp_france_winner
				set_global_flag = pfp_france_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed France + French Alps (P64) -> absorb Alps into France
	# Follow-up: only available after pfp_merge_france has fired
	# ----------------------------------------------------------------
	pfp_merge_france_alps = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_france_merged
			NOT = { has_global_flag = pfp_france_absorbed_alps }
			P64 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_france_alps_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_france_absorbed_alps }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_france_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_france }
			}
			P64 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_france_alps_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_france }
					save_global_event_target_as = pfp_france_country
					for_each_loop = {
						array = P64.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = P64  has_contested_owner = P64 } }
					add_core_of = event_target:pfp_france_country
					add_contested_owner = event_target:pfp_france_country
					remove_core_of = P64
					remove_claim_by = P64
					remove_contested_owner = P64
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_france }
					annex_country = { target = P64 transfer_troops = yes }
				}

				clear_global_event_target = pfp_france_country
				set_global_flag = pfp_france_absorbed_alps
			}
		}

		ai_will_do = { factor = 0 }
	}

}

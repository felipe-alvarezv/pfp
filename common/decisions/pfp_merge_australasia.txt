###########################################################################
# MERGE DECISIONS - AUSTRALASIA
# Allows the overlord to merge Australasian puppet states into a unified
# Australasia, then absorb additional Pacific puppets.
#
# Decision 1: pfp_merge_australasia
#   Requires Australia (R19), New Zealand (R20), and Papua New Guinea (R21).
#   The puppet with the most total factories becomes Australasia.
#
# Follow-up decisions (available after Australasia is formed):
#   pfp_merge_australasia_solomon_islands - Absorbs Solomon Islands (R22)
#   pfp_merge_australasia_bougainville    - Absorbs Bougainville (R23)
#   pfp_merge_australasia_new_caledonia   - Absorbs New Caledonia (R24)
#   pfp_merge_australasia_vanuatu         - Absorbs Vanuatu (R25)
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Australia (R19) + New Zealand (R20) + Papua New Guinea (R21) -> Australasia
	# ----------------------------------------------------------------
	pfp_merge_australasia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_australasia_merged }
			OR = {
				R19 = { is_subject_of = ROOT }
				R20 = { is_subject_of = ROOT }
				R21 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_australasia_not_yet_formed_tt
				NOT = { has_global_flag = pfp_australasia_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_australasia_puppets_ready_tt
				R19 = { is_subject_of = ROOT }
				R20 = { is_subject_of = ROOT }
				R21 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				R19 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				R20 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				R21 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				R19 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				R20 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				R21 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_australasia_effect_form
			custom_effect_tooltip = pfp_merge_australasia_effect_annex
			custom_effect_tooltip = pfp_merge_australasia_effect_cores
			hidden_effect = {
				R19 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_australasia_winner
				}
				R20 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_australasia_winner.pfp_fac } }
						save_global_event_target_as = pfp_australasia_winner
					}
				}
				R21 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_australasia_winner.pfp_fac } }
						save_global_event_target_as = pfp_australasia_winner
					}
				}

				if = {
					limit = { NOT = { event_target:pfp_australasia_winner = { tag = R19 } } }
					every_state = {
						limit = { OR = { is_owned_by = R19  has_contested_owner = R19 } }
						add_core_of = event_target:pfp_australasia_winner
						add_contested_owner = event_target:pfp_australasia_winner
						remove_core_of = R19
						remove_claim_by = R19
						remove_contested_owner = R19
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_australasia_winner = { tag = R20 } } }
					every_state = {
						limit = { OR = { is_owned_by = R20  has_contested_owner = R20 } }
						add_core_of = event_target:pfp_australasia_winner
						add_contested_owner = event_target:pfp_australasia_winner
						remove_core_of = R20
						remove_claim_by = R20
						remove_contested_owner = R20
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_australasia_winner = { tag = R21 } } }
					every_state = {
						limit = { OR = { is_owned_by = R21  has_contested_owner = R21 } }
						add_core_of = event_target:pfp_australasia_winner
						add_contested_owner = event_target:pfp_australasia_winner
						remove_core_of = R21
						remove_claim_by = R21
						remove_contested_owner = R21
					}
				}

				event_target:pfp_australasia_winner = {
					if = { limit = { NOT = { tag = R19 } } for_each_loop = { array = R19.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R20 } } for_each_loop = { array = R20.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R21 } } for_each_loop = { array = R21.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R19 } } annex_country = { target = R19 transfer_troops = yes } }
					if = { limit = { NOT = { tag = R20 } } annex_country = { target = R20 transfer_troops = yes } }
					if = { limit = { NOT = { tag = R21 } } annex_country = { target = R21 transfer_troops = yes } }
					set_cosmetic_tag = PFP_AUSTRALASIA
					set_country_flag = pfp_is_australasia
				}

				clear_global_event_target = pfp_australasia_winner
				set_global_flag = pfp_australasia_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

					limit = { has_country_flag = pfp_is_australasia }

					for_each_loop = { array = R22.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } }
				}
				every_state = {
					limit = { OR = { is_owned_by = R22  has_contested_owner = R22 } }

				}
				every_subject_country = {
					limit = { has_country_flag = pfp_is_australasia }
					annex_country = { target = R22 transfer_troops = yes }
				}
				clear_global_event_target = pfp_australasia_country
				set_global_flag = pfp_australasia_absorbed_solomon_islands
			}
		}

		ai_will_do = { factor = 0 }

	pfp_dutch_antilles = {
		allowed = {
			NOT = {
				original_tag = CRC #Netherlands Antilles
			}
		}
		available = {
			is_subject = no

			custom_trigger_tooltip = {
				tooltip = pfp_dutch_antilles_required_states_curacao
				has_full_control_of_state = 695 #Curacao
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 695 #Curacao
			}
			NOT = { has_global_flag = pfp_dutch_antilles_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 695 #Curacao
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_dutch_antilles_transfer_capital
			custom_effect_tooltip = pfp_dutch_antilles_puppet

				set_global_flag = pfp_dutch_antilles_flag
			}
		}
		ai_will_do = { factor = 0 }
	}

###########################################################################
# MERGE DECISIONS - KOREA
# Allows the overlord to merge North Korea and South Korea into a unified
# Korea.
#
# Decision 1: pfp_merge_korea
#   Requires North Korea (R38) and South Korea (R39) as PFP subjects.
#   The puppet with the most total factories becomes Korea.
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# North Korea (R38) + South Korea (R39) -> Korea
	# ----------------------------------------------------------------
	pfp_merge_korea = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_korea_merged }
			OR = {
				R38 = { is_subject_of = ROOT }
				R39 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_korea_not_yet_formed_tt
				NOT = { has_global_flag = pfp_korea_merged }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_korea_puppets_ready_tt
				R38 = { is_subject_of = ROOT }
				R39 = { is_subject_of = ROOT }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_subject_is_military_occupation_tt INDENT = "   " }
				R38 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
				R39 = { is_subject_of = ROOT  has_autonomy_state = pfp_autonomy_military_occupation }
			}
			custom_trigger_tooltip = {
				tooltip = { localization_key = pfp_merge_puppet_must_core_owned_territories_tt INDENT = "   " }
				R38 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
				R39 = { is_subject_of = ROOT  NOT = { any_state_in = { array = THIS.pfp_possible_cores  is_owned_by = PREV  NOT = { is_core_of = PREV } } } }
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_korea_effect_form
			custom_effect_tooltip = pfp_merge_korea_effect_annex
			custom_effect_tooltip = pfp_merge_korea_effect_cores
			hidden_effect = {
				R38 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					save_global_event_target_as = pfp_korea_winner
				}
				R39 = {
					set_variable = { pfp_fac = num_of_military_factories }
					add_to_variable = { pfp_fac = num_of_civilian_factories }
					if = {
						limit = { check_variable = { pfp_fac > event_target:pfp_korea_winner.pfp_fac } }
						save_global_event_target_as = pfp_korea_winner
					}
				}

				# Transfer state cores/ownership from loser to winner
				if = {
					limit = { NOT = { event_target:pfp_korea_winner = { tag = R38 } } }
					every_state = {
						limit = { OR = { is_owned_by = R38  has_contested_owner = R38 } }
						add_core_of = event_target:pfp_korea_winner
						add_contested_owner = event_target:pfp_korea_winner
						remove_core_of = R38
						remove_claim_by = R38
						remove_contested_owner = R38
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_korea_winner = { tag = R39 } } }
					every_state = {
						limit = { OR = { is_owned_by = R39  has_contested_owner = R39 } }
						add_core_of = event_target:pfp_korea_winner
						add_contested_owner = event_target:pfp_korea_winner
						remove_core_of = R39
						remove_claim_by = R39
						remove_contested_owner = R39
					}
				}

				# Transfer pfp_possible_cores arrays, annex loser, set tag and flag
				event_target:pfp_korea_winner = {
					if = { limit = { NOT = { tag = R38 } } for_each_loop = { array = R38.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R39 } } for_each_loop = { array = R39.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = R38 } } annex_country = { target = R38 transfer_troops = yes } }
					if = { limit = { NOT = { tag = R39 } } annex_country = { target = R39 transfer_troops = yes } }
					set_cosmetic_tag = PFP_KOREA
					set_country_flag = pfp_is_korea
				}

				clear_global_event_target = pfp_korea_winner
				set_global_flag = pfp_korea_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

}

pfp_category = {
	pfp_new_zealand = {
		allowed = {
			NOT = {
				original_tag = NZL #New Zealand
			}
		}
		available = {
			is_subject = no
			#Wellington
			custom_trigger_tooltip = {
				tooltip = pfp_new_zealand_required_states_wellington
				has_full_control_of_state = 284 #Wellington
			}
			#Auckland
			custom_trigger_tooltip = {
				tooltip = pfp_new_zealand_required_states_auckland
				has_full_control_of_state = 1079 #Auckland
			}
			#Marlborough
			custom_trigger_tooltip = {
				tooltip = pfp_new_zealand_required_states_marlborough
				has_full_control_of_state = 1080 #Marlborough
			}
			#Canterbury
			custom_trigger_tooltip = {
				tooltip = pfp_new_zealand_required_states_canterbury
				has_full_control_of_state = 723 #Canterbury
			}
			#Otago
			custom_trigger_tooltip = {
				tooltip = pfp_new_zealand_required_states_otago
				has_full_control_of_state = 1081 #Otago
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 284 #Wellington
				has_full_control_of_state = 1079 #Auckland
				has_full_control_of_state = 1080 #Marlborough
				has_full_control_of_state = 723 #Canterbury
				has_full_control_of_state = 1081 #Otago
			}
			NOT = { has_global_flag = pfp_new_zealand_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 284 #Wellington
				state = 1079 #Auckland
				state = 1080 #Marlborough
				state = 723 #Canterbury
				state = 1081 #Otago
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_new_zealand_transfer_capital
			custom_effect_tooltip = pfp_new_zealand_puppet
			custom_effect_tooltip = pfp_new_zealand_occupation_type
			custom_effect_tooltip = pfp_new_zealand_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {
				#Transfer control and ownership of capital to formable puppet
				284 = { transfer_state_to = R20 } #Wellington

				#Formable puppet becomes a puppet of ROOT
				puppet = R20

				#Code to assign generic portraits and set puppet color
				set_temp_variable = { pfp_portrait_tag = R20 }
				pfp_set_generic_portrait = yes
				pfp_set_autonomy_color = yes

				#Set puppet ideology and overlord flag for military occupations
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes
					if = {
						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_new_zealand = yes

				#Transfer all technology from ROOT to formable puppet
				R20 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_new_zealand_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

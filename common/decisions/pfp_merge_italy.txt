###########################################################################
# MERGE DECISIONS - ITALY
# Allows the overlord to merge Italian puppet states into a unified Italy.
#
# Decision 1: pfp_merge_italy
#   Requires Lombardy-Venetia (P92), Piedmont (P93), Tuscany (P95),
#   Papal States (P96), and Two Sicilies (P97) as PFP subjects.
#   The puppet with the most total factories + scaled manpower becomes
#   Italy (PFP_ITALY).
#
# Follow-up decision:
#   pfp_merge_italy_sardinia - Absorbs Sardinia (P94) into Italy.
###########################################################################

pfp_unification = {

	# ----------------------------------------------------------------
	# Lombardy-Venetia + Piedmont + Tuscany + Papal States + Two Sicilies
	# -> Italy
	# ----------------------------------------------------------------
	pfp_merge_italy = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			NOT = { has_global_flag = pfp_italy_merged }
			OR = {
				P92 = { is_subject_of = ROOT }
				P93 = { is_subject_of = ROOT }
				P95 = { is_subject_of = ROOT }
				P96 = { is_subject_of = ROOT }
				P97 = { is_subject_of = ROOT }
			}
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_italy_not_yet_formed_tt
				NOT = { has_global_flag = pfp_italy_merged }
			}
			P92 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P93 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P95 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P96 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
			P97 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
				custom_trigger_tooltip = {
					tooltip = pfp_merge_puppet_must_core_owned_territories_tt
					NOT = {
						any_state_in = {
							array = THIS.pfp_possible_cores
							is_owned_by = PREV
							NOT = { is_core_of = PREV }
						}
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_italy_effect_form
			custom_effect_tooltip = pfp_merge_italy_effect_annex
			custom_effect_tooltip = pfp_merge_italy_effect_cores
			hidden_effect = {
				# Score = mil factories + civ factories + (manpower / 100000)
				P92 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P93 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P95 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P96 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }
				P97 = { set_variable = { pfp_fac = num_of_military_factories } add_to_variable = { pfp_fac = num_of_civilian_factories } set_variable = { pfp_mp_temp = manpower } divide_variable = { pfp_mp_temp = 100000 } add_to_variable = { pfp_fac = pfp_mp_temp } }

				# Initialize winner with P92, then linear challenge
				P92 = { save_global_event_target_as = pfp_italy_winner }
				if = { limit = { check_variable = { P93.pfp_fac > event_target:pfp_italy_winner.pfp_fac } } P93 = { save_global_event_target_as = pfp_italy_winner } }
				if = { limit = { check_variable = { P95.pfp_fac > event_target:pfp_italy_winner.pfp_fac } } P95 = { save_global_event_target_as = pfp_italy_winner } }
				if = { limit = { check_variable = { P96.pfp_fac > event_target:pfp_italy_winner.pfp_fac } } P96 = { save_global_event_target_as = pfp_italy_winner } }
				if = { limit = { check_variable = { P97.pfp_fac > event_target:pfp_italy_winner.pfp_fac } } P97 = { save_global_event_target_as = pfp_italy_winner } }

				# Transfer state cores/ownership from non-winner subjects to winner
				if = {
					limit = { NOT = { event_target:pfp_italy_winner = { tag = P92 } } }
					every_state = {
						limit = { OR = { is_owned_by = P92  has_contested_owner = P92 } }
						add_core_of = event_target:pfp_italy_winner
						add_contested_owner = event_target:pfp_italy_winner
						remove_core_of = P92
						remove_claim_by = P92
						remove_contested_owner = P92
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_italy_winner = { tag = P93 } } }
					every_state = {
						limit = { OR = { is_owned_by = P93  has_contested_owner = P93 } }
						add_core_of = event_target:pfp_italy_winner
						add_contested_owner = event_target:pfp_italy_winner
						remove_core_of = P93
						remove_claim_by = P93
						remove_contested_owner = P93
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_italy_winner = { tag = P95 } } }
					every_state = {
						limit = { OR = { is_owned_by = P95  has_contested_owner = P95 } }
						add_core_of = event_target:pfp_italy_winner
						add_contested_owner = event_target:pfp_italy_winner
						remove_core_of = P95
						remove_claim_by = P95
						remove_contested_owner = P95
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_italy_winner = { tag = P96 } } }
					every_state = {
						limit = { OR = { is_owned_by = P96  has_contested_owner = P96 } }
						add_core_of = event_target:pfp_italy_winner
						add_contested_owner = event_target:pfp_italy_winner
						remove_core_of = P96
						remove_claim_by = P96
						remove_contested_owner = P96
					}
				}
				if = {
					limit = { NOT = { event_target:pfp_italy_winner = { tag = P97 } } }
					every_state = {
						limit = { OR = { is_owned_by = P97  has_contested_owner = P97 } }
						add_core_of = event_target:pfp_italy_winner
						add_contested_owner = event_target:pfp_italy_winner
						remove_core_of = P97
						remove_claim_by = P97
						remove_contested_owner = P97
					}
				}

				# Transfer pfp_possible_cores arrays, annex non-winners, set tag and flag
				event_target:pfp_italy_winner = {
					if = { limit = { NOT = { tag = P92 } } for_each_loop = { array = P92.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P93 } } for_each_loop = { array = P93.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P95 } } for_each_loop = { array = P95.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P96 } } for_each_loop = { array = P96.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P97 } } for_each_loop = { array = P97.pfp_possible_cores value = pfp_temp_core_id add_to_array = { pfp_possible_cores = pfp_temp_core_id } } }
					if = { limit = { NOT = { tag = P92 } } annex_country = { target = P92 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P93 } } annex_country = { target = P93 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P95 } } annex_country = { target = P95 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P96 } } annex_country = { target = P96 transfer_troops = yes } }
					if = { limit = { NOT = { tag = P97 } } annex_country = { target = P97 transfer_troops = yes } }
					set_cosmetic_tag = PFP_ITALY
					set_country_flag = pfp_is_italy
				}

				clear_global_event_target = pfp_italy_winner
				set_global_flag = pfp_italy_merged
			}
		}

		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------
	# Formed Italy + Sardinia (P94) -> absorbed into Italy
	# ----------------------------------------------------------------
	pfp_merge_italy_sardinia = {
		icon = generic_nationalism

		visible = {
			is_subject = no
			has_global_flag = pfp_italy_merged
			NOT = { has_global_flag = pfp_italy_absorbed_sardinia }
			P94 = { is_subject_of = ROOT }
		}

		available = {
			is_ai = no
			custom_trigger_tooltip = {
				tooltip = pfp_merge_italy_sardinia_not_yet_absorbed_tt
				NOT = { has_global_flag = pfp_italy_absorbed_sardinia }
			}
			custom_trigger_tooltip = {
				tooltip = pfp_merge_italy_is_formed_tt
				any_subject_country = { has_country_flag = pfp_is_italy }
			}
			P94 = {
				is_subject_of = ROOT
				custom_trigger_tooltip = {
					tooltip = pfp_merge_subject_is_military_occupation_tt
					OR = {
						has_autonomy_state = pfp_autonomy_military_occupation
						has_autonomy_state = pfp_autonomy_military_occupation_painted
					}
				}
			}
		}

		fire_only_once = yes

		complete_effect = {
			custom_effect_tooltip = pfp_merge_italy_sardinia_effect_annex
			hidden_effect = {
				every_subject_country = {
					limit = { has_country_flag = pfp_is_italy }
					save_global_event_target_as = pfp_italy_country
					for_each_loop = {
						array = P94.pfp_possible_cores
						value = pfp_temp_core_id
						add_to_array = { pfp_possible_cores = pfp_temp_core_id }
					}
				}

				every_state = {
					limit = { OR = { is_owned_by = P94  has_contested_owner = P94 } }
					add_core_of = event_target:pfp_italy_country
					add_contested_owner = event_target:pfp_italy_country
					remove_core_of = P94
					remove_claim_by = P94
					remove_contested_owner = P94
				}

				every_subject_country = {
					limit = { has_country_flag = pfp_is_italy }
					annex_country = { target = P94 transfer_troops = yes }
				}

				clear_global_event_target = pfp_italy_country
				set_global_flag = pfp_italy_absorbed_sardinia
			}
		}

		ai_will_do = { factor = 0 }
	}

}

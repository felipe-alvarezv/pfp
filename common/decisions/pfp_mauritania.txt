pfp_category = {
	pfp_mauritania = {
		allowed = {
			NOT = {
				original_tag = MRT #Mauritania
			}
		}
		available = {
			is_subject = no
			#Mauritania
			custom_trigger_tooltip = {
				tooltip = pfp_mauritania_required_states_mauritania
				has_full_control_of_state = 557 #Mauritania
			}
			custom_trigger_tooltip = {
				tooltip = pfp_mauritania_required_states_mauritanian_desert
				has_full_control_of_state = 786 #Mauritanian Desert
			}
		}
		visible = {
			OR = {
				has_full_control_of_state = 557 #Mauritania
				has_full_control_of_state = 786 #Mauritanian Desert
			}
			NOT = { has_global_flag = pfp_mauritania_flag }
		}
		highlight_states = {
			highlight_state_targets = {
				state = 557 #Mauritania
				state = 786 #Mauritanian Desert
			}
		}
		icon = generic_military
		complete_effect = {
			custom_effect_tooltip = pfp_mauritania_transfer_capital
			custom_effect_tooltip = pfp_mauritania_puppet
			custom_effect_tooltip = pfp_mauritania_occupation_type
			custom_effect_tooltip = pfp_mauritania_state_controller

			#Create formable puppet with all its states and then puppet
			hidden_effect = {

				557 = { transfer_state_to = Q52 } #Mauritania

				pfp_set_autonomy_color = yes
				var:pfp_portrait_tag = {
					pfp_set_puppet_ideology = yes

						limit = {
							OR = {
								has_autonomy_state = pfp_autonomy_military_occupation
								has_autonomy_state = pfp_autonomy_military_occupation_painted
							}
						}
						pfp_set_use_overlord_flag = yes
					}
				}

				#Formable puppet will core or contest its designated territory
				pfp_release_mauritania = yes

				#Transfer all technology from ROOT to formable puppet
				Q52 = { pfp_transfer_technology = yes }

				#Set global flag to prevent future decision use
				set_global_flag = pfp_mauritania_flag
			}
		}
		ai_will_do = { factor = 0 }
	}
}

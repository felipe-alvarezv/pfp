on_actions = {
    #Pipe's Formable Puppets - Ruling Party Change
    on_ruling_party_change = {
        effect = {
            every_subject_country = {
                limit = {
                    OR = {
                        has_autonomy_state = pfp_autonomy_military_occupation
                        has_autonomy_state = pfp_autonomy_military_occupation_painted
                    }
                }

                if = {
                    limit = { NOT = { has_government = var:overlord } }
                    set_politics = {
                        ruling_party = var:overlord
                    }
                    if = {
                        limit = { has_government = democratic }
                        set_popularities = {
                            democratic = 100
                            fascism = 0
                            communism = 0
                            neutrality = 0
                        }
                    }
                    else_if = {
                        limit = { has_government = fascism }
                        set_popularities = {
                            democratic = 0
                            fascism = 100
                            communism = 0
                            neutrality = 0
                        }
                    }
                    else_if = {
                        limit = { has_government = communism }
                        set_popularities = {
                            democratic = 0
                            fascism = 0
                            communism = 100
                            neutrality = 0
                        }
                    }
                    else_if = {
                        limit = { has_government = neutrality }
                        set_popularities = {
                            democratic = 0
                            fascism = 0
                            communism = 0
                            neutrality = 100
                        }
                    }
                }

                #Set puppet flag according to selected setting
                pfp_set_use_overlord_flag = yes
            }
        }
    }
}
